            <h2 class="text-xl font-bold text-indigo-400 border-b border-slate-700 pb-2">Theory: Higher-Order Functions in Python</h2>
            <p class="text-sm text-slate-300 leading-relaxed">
                A <strong>Higher-Order Function</strong> is a function that either accepts one or more functions as arguments or returns a function as its result. In Python, functions are first-class citizens, meaning they can be assigned to variables, passed into other functions, or returned. Built-in higher-order functions like <code class="bg-slate-900 text-amber-300 px-1.5 py-0.5 rounded font-mono">map()</code>, <code class="bg-slate-900 text-amber-300 px-1.5 py-0.5 rounded font-mono">filter()</code>, and <code class="bg-slate-900 text-amber-300 px-1.5 py-0.5 rounded font-mono">reduce()</code> enable clean functional programming paradigms.
            </p>

            <div class="grid grid-cols-1 md:grid-cols-3 gap-4 text-sm">
                
                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-arrows-split-up-and-left text-emerald-400"></i> map(func, iterable)</span>
                        <span class="text-[10px] bg-emerald-500/20 text-emerald-300 px-2 py-0.5 rounded font-sans">Transformation</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Applies a given function to every item of an iterable and returns an iterator: <code class="text-amber-300">map(lambda x: x*2, nums)</code>.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-filter text-amber-400"></i> filter(pred, iterable)</span>
                        <span class="text-[10px] bg-amber-500/20 text-amber-300 px-2 py-0.5 rounded font-sans">Selection</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Filters items from an iterable for which a boolean predicate returns <code class="text-emerald-300">True</code>: <code class="text-amber-300">filter(lambda x: x%2==0, nums)</code>.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-compress text-cyan-400"></i> reduce(func, iterable)</span>
                        <span class="text-[10px] bg-cyan-500/20 text-cyan-300 px-2 py-0.5 rounded font-sans">functools</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Applies a binary function cumulatively to iterable items from left to right to reduce the sequence to a single value: <code class="text-amber-300">reduce(lambda a, b: a+b, nums)</code>.
                    </p>
                </div>

            </div>

            <div class="bg-slate-900/60 p-4 rounded-lg border border-slate-700 text-xs text-slate-300 space-y-2">
                <h4 class="font-bold text-indigo-300 text-sm"><i class="fa-solid fa-bolt text-amber-300"></i> Anonymous Lambda Functions</h4>
                <p class="text-slate-400">
                    A <code class="text-emerald-300 font-mono">lambda</code> expression creates inline, anonymous single-expression functions without formal <code class="text-emerald-300 font-mono">def</code> blocks. Syntax: <code class="text-amber-300 font-mono">lambda arg1, arg2: expression</code>.
                </p>
            </div>
        </div>

        function startSimulation() {
            updateScenarioDisplay();
            buildExecutionSteps();
            currentStepIdx = 0;

            hideResultBanner();
            clearConsole();

            const btnStep = document.getElementById('btnStep');
            btnStep.disabled = false;
            btnStep.className = "bg-emerald-600 hover:bg-emerald-500 text-white font-medium px-4 py-2 rounded-lg text-sm transition active:scale-95 shadow-lg shadow-emerald-600/20";

            appendConsole(`>>> Executing: higher_order_funcs.py [Scenario: ${scenario}]`);
            executeStepAtIndex(0);
        }

        function executeStepAtIndex(idx) {
            const step = stepList[idx];
            highlightLine(step.line);

            document.getElementById('lblVar1').innerText = step.v1.label;
            document.getElementById('valVar1').innerText = step.v1.val;
            document.getElementById('typeVar1').innerText = step.v1.type;

            document.getElementById('lblVar2').innerText = step.v2.label;
            document.getElementById('valVar2').innerText = step.v2.val;
            document.getElementById('typeVar2').innerText = step.v2.type;

            document.getElementById('lblVar3').innerText = step.v3.label;
            document.getElementById('valVar3').innerText = step.v3.val;
            document.getElementById('typeVar3').innerText = step.v3.type;

            triggerMemUpdate('memVar1');
            triggerMemUpdate('memVar2');
            triggerMemUpdate('memVar3');

            appendConsole(step.consoleMsg);
            document.getElementById('explanationText').innerText = step.explanation;
            document.getElementById('currentStepBadge').innerText = `Step ${idx + 1}/${stepList.length}`;

            if (idx === stepList.length - 1) {
                showResultBanner("Higher-Order Function Execution Completed", false);
                const btnStep = document.getElementById('btnStep');
                btnStep.disabled = true;
                btnStep.className = "bg-slate-700 text-slate-400 font-medium px-4 py-2 rounded-lg text-sm transition cursor-not-allowed";
                document.getElementById('currentStepBadge').innerText = "Finished";
            }
        }

        function stepExecute() {
            if (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function runAll() {
            startSimulation();
            while (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function sendAiQuery() {
            const input = document.getElementById('aiQueryInput');
            const query = input.value.trim();
            if (!query) return;

            const chatBox = document.getElementById('aiChatBox');

            const userDiv = document.createElement('div');
            userDiv.className = 'flex items-start gap-3 justify-end';
            userDiv.innerHTML = `
                <div class="bg-indigo-600/30 border border-indigo-500/40 p-3 rounded-lg text-slate-200">
                    ${query}
                </div>
                <div class="w-7 h-7 rounded-full bg-slate-700 flex items-center justify-center text-slate-300 text-xs shrink-0">You</div>
            `;
            chatBox.appendChild(userDiv);
            input.value = '';

            let response = "Try testing map(), filter(), and reduce() scenarios in the simulator to see how functions operate on lists!";
            const qLower = query.toLowerCase();
            if (qLower.includes('map') || qLower.includes('transform')) {
                response = "`map(func, iterable)` applies `func` to every item in the list and returns an iterator of transformed values!";
            } else if (qLower.includes('filter') || qLower.includes('predicate')) {
                response = "`filter(predicate, iterable)` tests each element against a boolean function and keeps only the elements where `predicate(x)` is `True`!";
            } else if (qLower.includes('reduce') || qLower.includes('functools') || qLower.includes('cumulative')) {
                response = "`reduce(func, iterable)` from `functools` applies a binary function cumulatively across items to collapse a list into a single value (e.g. cumulative sum or product)!";
            } else if (qLower.includes('lambda') || qLower.includes('anonymous')) {
                response = "A `lambda` function is a short, anonymous function defined inline without `def`: e.g. `lambda x: x * 2`!";
            }

            setTimeout(() => {
                const aiDiv = document.createElement('div');
                aiDiv.className = 'flex items-start gap-3';
                aiDiv.innerHTML = `
                    <div class="w-7 h-7 rounded-full bg-indigo-600 flex items-center justify-center text-white text-xs shrink-0">AI</div>
                    <div class="bg-slate-800 p-3 rounded-lg border border-slate-700 text-slate-200">
                        ${response}
                    </div>
                `;
                chatBox.appendChild(aiDiv);
                chatBox.scrollTop = chatBox.scrollHeight;
            }, 500);
        }
