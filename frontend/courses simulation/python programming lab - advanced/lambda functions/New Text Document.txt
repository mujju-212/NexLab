            <h2 class="text-xl font-bold text-indigo-400 border-b border-slate-700 pb-2">Theory: Anonymous Lambda Functions in Python</h2>
            <p class="text-sm text-slate-300 leading-relaxed">
                A <strong>Lambda Function</strong> in Python is an inline, anonymous function defined without using the standard <code class="bg-slate-900 text-amber-300 px-1.5 py-0.5 rounded font-mono">def</code> keyword. Lambda functions can take any number of arguments but can only evaluate a single expression and implicitly return its result.
            </p>

            <div class="grid grid-cols-1 md:grid-cols-3 gap-4 text-sm">
                
                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-bolt text-emerald-400"></i> Single-Argument Syntax</span>
                        <span class="text-[10px] bg-emerald-500/20 text-emerald-300 px-2 py-0.5 rounded font-sans">Basic</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Syntax: <code class="text-amber-300">lambda x: x ** 2</code>. Takes one input argument <code class="text-emerald-300">x</code> and returns its square.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-calculator text-amber-400"></i> Multi-Argument Syntax</span>
                        <span class="text-[10px] bg-amber-500/20 text-amber-300 px-2 py-0.5 rounded font-sans">Multiple Inputs</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Syntax: <code class="text-amber-300">lambda a, b: a + b</code>. Accepts multiple comma-separated arguments for evaluation.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-code-branch text-cyan-400"></i> Conditional Lambda</span>
                        <span class="text-[10px] bg-cyan-500/20 text-cyan-300 px-2 py-0.5 rounded font-sans">Ternary</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Syntax: <code class="text-amber-300">lambda a, b: a if a > b else b</code>. Evaluates ternary inline conditional expressions.
                    </p>
                </div>

            </div>

            <div class="bg-slate-900/60 p-4 rounded-lg border border-slate-700 text-xs text-slate-300 space-y-2">
                <h4 class="font-bold text-indigo-300 text-sm"><i class="fa-solid fa-circle-info text-amber-300"></i> When to Use Lambda Functions</h4>
                <ul class="list-disc list-inside space-y-1 text-slate-400">
                    <li>For short, throwaway functions needed briefly in code (e.g. key functions in <code class="text-emerald-300">sort()</code> or <code class="text-emerald-300">sorted()</code>).</li>
                    <li>Inside higher-order functions like <code class="text-emerald-300">map()</code> and <code class="text-emerald-300">filter()</code>.</li>
                    <li>Avoid complex logic inside lambdas; if multiple statements or loops are needed, use a formal <code class="text-emerald-300">def</code> block!</li>
                </ul>
            </div>
        </div>

        function startSimulation() {
            updateScenarioDisplay();
            buildExecutionSteps();
            currentStepIdx = 0;

            hideResultBanner();
            clearConsole();

            const btnStep = document.getElementById('btnStep');
            btnStep.disabled = false;
            btnStep.className = "bg-emerald-600 hover:bg-emerald-500 text-white font-medium px-4 py-2 rounded-lg text-sm transition active:scale-95 shadow-lg shadow-emerald-600/20";

            appendConsole(`>>> Executing: lambda_demo.py [Scenario: ${scenario}]`);
            executeStepAtIndex(0);
        }

        function executeStepAtIndex(idx) {
            const step = stepList[idx];
            highlightLine(step.line);

            document.getElementById('lblVar1').innerText = step.v1.label;
            document.getElementById('valVar1').innerText = step.v1.val;
            document.getElementById('typeVar1').innerText = step.v1.type;

            document.getElementById('lblVar2').innerText = step.v2.label;
            document.getElementById('valVar2').innerText = step.v2.val;
            document.getElementById('typeVar2').innerText = step.v2.type;

            document.getElementById('lblVar3').innerText = step.v3.label;
            document.getElementById('valVar3').innerText = step.v3.val;
            document.getElementById('typeVar3').innerText = step.v3.type;

            triggerMemUpdate('memVar1');
            triggerMemUpdate('memVar2');
            triggerMemUpdate('memVar3');

            appendConsole(step.consoleMsg);
            document.getElementById('explanationText').innerText = step.explanation;
            document.getElementById('currentStepBadge').innerText = `Step ${idx + 1}/${stepList.length}`;

            if (idx === stepList.length - 1) {
                showResultBanner("Lambda Function Execution Completed", false);
                const btnStep = document.getElementById('btnStep');
                btnStep.disabled = true;
                btnStep.className = "bg-slate-700 text-slate-400 font-medium px-4 py-2 rounded-lg text-sm transition cursor-not-allowed";
                document.getElementById('currentStepBadge').innerText = "Finished";
            }
        }

        function stepExecute() {
            if (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function runAll() {
            startSimulation();
            while (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function resetSimulation() {
            currentStepIdx = 0;
            highlightLine(0);
            clearConsole();
            hideResultBanner();
            appendConsole("<p class='text-slate-500 italic'>>>> Click 'Run / Execute All' or 'Start Simulation' to execute.</p>");
            document.getElementById('explanationText').innerText = "Select parameters and click 'Start Simulation' or 'Run / Execute All' to execute.";
            document.getElementById('currentStepBadge').innerText = "Step: Ready";

            const btnStep = document.getElementById('btnStep');
            btnStep.disabled = true;
            btnStep.className = "bg-slate-700 text-slate-400 font-medium px-4 py-2 rounded-lg text-sm transition cursor-not-allowed";
        }

        function highlightLine(lineNum) {
            document.querySelectorAll('.code-line').forEach(el => el.classList.remove('active'));
            if (lineNum > 0) {
                const target = document.getElementById(`line${lineNum}`);
                if (target) target.classList.add('active');
            }
        }

        function clearConsole() {
            document.getElementById('terminal').innerHTML = '';
        }

        function appendConsole(htmlContent) {
            const term = document.getElementById('terminal');
            const p = document.createElement('p');
            p.innerHTML = htmlContent;
            term.appendChild(p);
            term.scrollTop = term.scrollHeight;
        }

        function triggerMemUpdate(memId) {
            const el = document.getElementById(memId);
            if (!el) return;
            el.classList.add('updated');
            setTimeout(() => { el.classList.remove('updated'); }, 500);
        }

        function checkPretest() {
            const q1 = document.querySelector('input[name="pq1"]:checked')?.value;
            const q2 = document.querySelector('input[name="pq2"]:checked')?.value;
            let score = 0;

            const q1Box = document.getElementById('preq1-box');
            const q1fb = document.getElementById('preq1-feedback');
            q1fb.classList.remove('hidden');
            if (q1 === 'b') {
                score++;
                q1Box.className = "bg-slate-900 p-4 rounded-lg border border-emerald-500/50";
                q1fb.innerText = "✓ Correct! `lambda` is used to create anonymous inline functions.";
                q1fb.className = "text-xs mt-2 font-medium text-emerald-400";
            } else {
                q1Box.className = "bg-slate-900 p-4 rounded-lg border border-rose-500/50";
                q1fb.innerText = "✗ Incorrect. Use the `lambda` keyword.";
                q1fb.className = "text-xs mt-2 font-medium text-rose-400";
            }

            const q2Box = document.getElementById('preq2-box');
            const q2fb = document.getElementById('preq2-feedback');
            q2fb.classList.remove('hidden');
            if (q2 === 'a') {
                score++;
                q2Box.className = "bg-slate-900 p-4 rounded-lg border border-emerald-500/50";
                q2fb.innerText = "✓ Correct! A lambda function can only contain a single expression.";
                q2fb.className = "text-xs mt-2 font-medium text-emerald-400";
            } else {
                q2Box.className = "bg-slate-900 p-4 rounded-lg border border-rose-500/50";
                q2fb.innerText = "✗ Incorrect. Lambdas are limited to a single expression.";
                q2fb.className = "text-xs mt-2 font-medium text-rose-400";
            }

            const badge = document.getElementById('pretest-score');
            badge.innerText = `Score: ${score} / 2`;
            badge.classList.remove('hidden');
        }

                q1Box.className = "bg-slate-900 p-4 rounded-lg border border-emerald-500/50";

        function sendAiQuery() {
            const input = document.getElementById('aiQueryInput');
            const query = input.value.trim();
            if (!query) return;

            const chatBox = document.getElementById('aiChatBox');

            const userDiv = document.createElement('div');
            userDiv.className = 'flex items-start gap-3 justify-end';
            userDiv.innerHTML = `
                <div class="bg-indigo-600/30 border border-indigo-500/40 p-3 rounded-lg text-slate-200">
                    ${query}
                </div>
                <div class="w-7 h-7 rounded-full bg-slate-700 flex items-center justify-center text-slate-300 text-xs shrink-0">You</div>
            `;
            chatBox.appendChild(userDiv);
            input.value = '';

            let response = "Try testing lambda functions in the simulator to see how inline anonymous functions execute!";
            const qLower = query.toLowerCase();
            if (qLower.includes('def') || qLower.includes('difference')) {
                response = "`def` creates a formal, named function with multiple statements. `lambda` creates an inline, anonymous single-expression function!";
            } else if (qLower.includes('syntax') || qLower.includes('how')) {
                response = "The syntax is `lambda arg1, arg2: expression`. It implicitly returns the result of `expression` without an explicit `return` keyword!";
            } else if (qLower.includes('sort') || qLower.includes('key')) {
                response = "Lambdas are frequently passed as `key` functions in `sorted()` or `list.sort()`—for example `students.sort(key=lambda s: s['mark'])`!";
            }

            setTimeout(() => {
                const aiDiv = document.createElement('div');
                aiDiv.className = 'flex items-start gap-3';
                aiDiv.innerHTML = `
                    <div class="w-7 h-7 rounded-full bg-indigo-600 flex items-center justify-center text-white text-xs shrink-0">AI</div>
                    <div class="bg-slate-800 p-3 rounded-lg border border-slate-700 text-slate-200">
                        ${response}
                    </div>
                `;
                chatBox.appendChild(aiDiv);
                chatBox.scrollTop = chatBox.scrollHeight;
            }, 500);
        }
