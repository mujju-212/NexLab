        <div id="content-theory" class="tab-content bg-slate-800 border border-slate-700 rounded-xl p-6 shadow-md space-y-6">
            <h2 class="text-xl font-bold text-indigo-400 border-b border-slate-700 pb-2">Theory: Extended Formal Argument Syntax in Python</h2>
            <p class="text-sm text-slate-300 leading-relaxed">
                Python allows functions to accept an arbitrary number of arguments using <strong>Extended Formal Argument Syntax</strong>. The single asterisk syntax (<code class="bg-slate-900 text-amber-300 px-1.5 py-0.5 rounded font-mono">*args</code>) captures extra positional arguments into a <strong>tuple</strong>, while the double asterisk syntax (<code class="bg-slate-900 text-amber-300 px-1.5 py-0.5 rounded font-mono">**kwargs</code>) captures keyword arguments into a <strong>dictionary</strong>.
            </p>

            <div class="grid grid-cols-1 md:grid-cols-3 gap-4 text-sm">
                
                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-asterisk text-emerald-400"></i> *args Syntax</span>
                        <span class="text-[10px] bg-emerald-500/20 text-emerald-300 px-2 py-0.5 rounded font-sans">Positional Tuple</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Packs variable non-keyword arguments into an immutable tuple: <code class="text-amber-300">def add(*args): ...</code>. Allows functions to process any quantity of positional parameters.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-key text-amber-400"></i> **kwargs Syntax</span>
                        <span class="text-[10px] bg-amber-500/20 text-amber-300 px-2 py-0.5 rounded font-sans">Keyword Dict</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Packs variable named keyword arguments into a dictionary: <code class="text-amber-300">def info(**kwargs): ...</code>. Access parameters using key lookups.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-sliders text-cyan-400"></i> Parameter Order</span>
                        <span class="text-[10px] bg-cyan-500/20 text-cyan-300 px-2 py-0.5 rounded font-sans">Sequence Rule</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Strict declaration sequence: Standard positional arguments, then <code class="text-amber-300">*args</code>, default parameters, and finally <code class="text-amber-300">**kwargs</code>.
                    </p>
                </div>

            </div>

            <div class="bg-slate-900/60 p-4 rounded-lg border border-slate-700 text-xs text-slate-300 space-y-2">
                <h4 class="font-bold text-indigo-300 text-sm"><i class="fa-solid fa-circle-info text-amber-300"></i> Key Syntax Comparison</h4>
                <ul class="list-disc list-inside space-y-1 text-slate-400 font-mono">
                    <li><strong class="text-slate-200 font-sans">*args:</strong> <code class="text-emerald-300">func(1, 2, 3)</code> $\rightarrow$ <code class="text-slate-300">args = (1, 2, 3)</code> (type: <code class="text-amber-300">tuple</code>)</li>
                    <li><strong class="text-slate-200 font-sans">**kwargs:</strong> <code class="text-emerald-300">func(a=1, b=2)</code> $\rightarrow$ <code class="text-slate-300">kwargs = {'a': 1, 'b': 2}</code> (type: <code class="text-amber-300">dict</code>)</li>
                    <li><strong class="text-slate-200 font-sans">Unpacking:</strong> <code class="text-emerald-300">func(*my_list)</code> or <code class="text-emerald-300">func(**my_dict)</code> unpacks containers directly into call arguments.</li>
                </ul>
            </div>
        </div>

        function startSimulation() {
            updateScenarioDisplay();
            buildExecutionSteps();
            currentStepIdx = 0;

            hideResultBanner();
            clearConsole();

            const btnStep = document.getElementById('btnStep');
            btnStep.disabled = false;

            appendConsole(`>>> Executing: extended_args.py [Scenario: ${scenario}]`);
            executeStepAtIndex(0);
        }

        function executeStepAtIndex(idx) {
            const step = stepList[idx];
            highlightLine(step.line);

            document.getElementById('lblVar1').innerText = step.v1.label;
            document.getElementById('valVar1').innerText = step.v1.val;
            document.getElementById('typeVar1').innerText = step.v1.type;

            document.getElementById('lblVar2').innerText = step.v2.label;
            document.getElementById('valVar2').innerText = step.v2.val;
            document.getElementById('typeVar2').innerText = step.v2.type;

            document.getElementById('lblVar3').innerText = step.v3.label;
            document.getElementById('valVar3').innerText = step.v3.val;
            document.getElementById('typeVar3').innerText = step.v3.type;

            triggerMemUpdate('memVar1');
            triggerMemUpdate('memVar2');
            triggerMemUpdate('memVar3');

            appendConsole(step.consoleMsg);
            document.getElementById('explanationText').innerText = step.explanation;
            document.getElementById('currentStepBadge').innerText = `Step ${idx + 1}/${stepList.length}`;

            if (idx === stepList.length - 1) {
                showResultBanner("Extended Arguments Execution Completed", false);
                const btnStep = document.getElementById('btnStep');
                btnStep.disabled = true;
                btnStep.className = "bg-slate-700 text-slate-400 font-medium px-4 py-2 rounded-lg text-sm transition cursor-not-allowed";
                document.getElementById('currentStepBadge').innerText = "Finished";
            }
        }

        function stepExecute() {
            if (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function runAll() {
            startSimulation();
            while (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function sendAiQuery() {
            const input = document.getElementById('aiQueryInput');
            const query = input.value.trim();
            if (!query) return;

            const chatBox = document.getElementById('aiChatBox');

            const userDiv = document.createElement('div');
            userDiv.className = 'flex items-start gap-3 justify-end';
            userDiv.innerHTML = `
                <div class="bg-indigo-600/30 border border-indigo-500/40 p-3 rounded-lg text-slate-200">
                    ${query}
                </div>
                <div class="w-7 h-7 rounded-full bg-slate-700 flex items-center justify-center text-slate-300 text-xs shrink-0">You</div>
            `;
            chatBox.appendChild(userDiv);
            input.value = '';

            let response = "Try running extended argument scenarios in the simulator to see how *args tuples and **kwargs dicts update in memory!";
            const qLower = query.toLowerCase();
            if (qLower.includes('args') && !qLower.includes('kwargs')) {
                response = "`*args` allows a function to accept any number of positional arguments, which are stored as a tuple inside the function!";
            } else if (qLower.includes('kwargs')) {
                response = "`**kwargs` captures arbitrary keyword arguments as a dictionary (e.g., `func(name='Alice', age=20)` creates `kwargs = {'name': 'Alice', 'age': 20}`)!";
            } else if (qLower.includes('order') || qLower.includes('sequence')) {
                response = "The parameter declaration sequence must follow: 1. Positional args, 2. `*args`, 3. Default parameters, 4. `**kwargs`!";
            } else if (qLower.includes('unpack')) {
                response = "You can unpack containers into function arguments using `*list` for positional args or `**dict` for keyword args!";
            }

            setTimeout(() => {
                const aiDiv = document.createElement('div');
                aiDiv.className = 'flex items-start gap-3';
                aiDiv.innerHTML = `
                    <div class="w-7 h-7 rounded-full bg-indigo-600 flex items-center justify-center text-white text-xs shrink-0">AI</div>
                    <div class="bg-slate-800 p-3 rounded-lg border border-slate-700 text-slate-200">
                        ${response}
                    </div>
                `;
                chatBox.appendChild(aiDiv);
                chatBox.scrollTop = chatBox.scrollHeight;
            }, 500);
        }
