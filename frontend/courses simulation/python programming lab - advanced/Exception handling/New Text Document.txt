        <div id="content-theory" class="tab-content bg-slate-800 border border-slate-700 rounded-xl p-6 shadow-md space-y-6">
            <h2 class="text-xl font-bold text-indigo-400 border-b border-slate-700 pb-2">Theory: Exception Handling in Python</h2>
            <p class="text-sm text-slate-300 leading-relaxed">
                An <strong>Exception</strong> is an event or error that occurs during program execution, interrupting the normal flow of instructions. Python handles runtime errors gracefully using the <code class="bg-slate-900 text-amber-300 px-1.5 py-0.5 rounded font-mono">try-except</code> block construct, preventing sudden script crashes.
            </p>

            <div class="grid grid-cols-1 md:grid-cols-3 gap-4 text-sm">
                
                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-shield-cat text-emerald-400"></i> try ... except</span>
                        <span class="text-[10px] bg-emerald-500/20 text-emerald-300 px-2 py-0.5 rounded font-sans">Basic Catch</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Code that might throw a runtime error is placed inside the <code class="text-amber-300">try</code> block. If an error occurs, execution jumps to the matching <code class="text-amber-300">except</code> block.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-code-merge text-amber-400"></i> else & finally</span>
                        <span class="text-[10px] bg-amber-500/20 text-amber-300 px-2 py-0.5 rounded font-sans">Clauses</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        <code class="text-amber-300">else</code> runs ONLY if NO exceptions occur in the <code class="text-amber-300">try</code> block. <code class="text-amber-300">finally</code> runs ALWAYS, regardless of errors (useful for resource cleanup).
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-triangle-exclamation text-rose-400"></i> raise Exception</span>
                        <span class="text-[10px] bg-rose-500/20 text-rose-300 px-2 py-0.5 rounded font-sans">Manual Trigger</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Developers can explicitly trigger errors using the <code class="text-rose-400">raise</code> statement (e.g., <code class="text-rose-400">raise ValueError("Invalid Input")</code>) for business logic validation.
                    </p>
                </div>

            </div>

            <div class="bg-slate-900/60 p-4 rounded-lg border border-slate-700 text-xs text-slate-300 space-y-2">
                <h4 class="font-bold text-indigo-300 text-sm"><i class="fa-solid fa-circle-info text-amber-300"></i> Common Standard Exception Types</h4>
                <ul class="list-disc list-inside space-y-1 text-slate-400 font-mono">
                    <li><strong class="text-slate-200 font-sans">ZeroDivisionError:</strong> Raised when dividing a number by zero (<code class="text-emerald-300">10 / 0</code>).</li>
                    <li><strong class="text-slate-200 font-sans">ValueError:</strong> Raised when a function receives an argument of right type but inappropriate value (<code class="text-emerald-300">int("abc")</code>).</li>
                    <li><strong class="text-slate-200 font-sans">TypeError:</strong> Raised when an operation is performed on an inappropriate data type.</li>
                </ul>
            </div>
        </div>

        function stepExecute() {
            if (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }

            startSimulation();
            while (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function sendAiQuery() {
            const input = document.getElementById('aiQueryInput');
            const query = input.value.trim();
            if (!query) return;

            const chatBox = document.getElementById('aiChatBox');

            const userDiv = document.createElement('div');
            userDiv.className = 'flex items-start gap-3 justify-end';
            userDiv.innerHTML = `
                <div class="bg-indigo-600/30 border border-indigo-500/40 p-3 rounded-lg text-slate-200">
                    ${query}
                </div>
                <div class="w-7 h-7 rounded-full bg-slate-700 flex items-center justify-center text-slate-300 text-xs shrink-0">You</div>
            `;
            chatBox.appendChild(userDiv);
            input.value = '';

            let response = "Try running exception handling scenarios in the simulator to observe how try-except blocks catch errors!";
            const qLower = query.toLowerCase();
            if (qLower.includes('finally') || qLower.includes('else')) {
                response = "`else` executes ONLY if no exception occurred in `try`. `finally` executes ALWAYS, making it essential for releasing resources (closing files, DB connections)!";
            } else if (qLower.includes('raise') || qLower.includes('custom') || qLower.includes('trigger')) {
                response = "Use `raise ExceptionType('message')` to trigger an error manually when validation conditions fail!";
            } else if (qLower.includes('zerodivision') || qLower.includes('valueerror') || qLower.includes('catch')) {
                response = "Catch specific error types (`except ZeroDivisionError:`, `except ValueError:`) to handle distinct runtime errors safely!";
            }

            setTimeout(() => {
                const aiDiv = document.createElement('div');
                aiDiv.className = 'flex items-start gap-3';
                aiDiv.innerHTML = `
                    <div class="w-7 h-7 rounded-full bg-indigo-600 flex items-center justify-center text-white text-xs shrink-0">AI</div>
                    <div class="bg-slate-800 p-3 rounded-lg border border-slate-700 text-slate-200">
                        ${response}
                    </div>
