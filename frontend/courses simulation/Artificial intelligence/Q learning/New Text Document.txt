    <script>
        let qTable = Array(9).fill().map(() => ({up:0, down:0, left:0, right:0}));
        let episodes = 0;

        function switchTab(t) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('text-indigo-400', 'bg-slate-700/60', 'border', 'border-slate-600'));
            document.getElementById('content-' + t).classList.remove('hidden');
            const activeBtn = Array.from(document.querySelectorAll('.tab-btn')).find(b => b.innerText.toLowerCase().includes(t));
            if (activeBtn) activeBtn.classList.add('text-indigo-400', 'bg-slate-700/60', 'border', 'border-slate-600');
        }

        function renderTable() {
            const container = document.getElementById('qTable');
            container.innerHTML = '';
            qTable.forEach((cell, i) => {
                container.innerHTML += `
                    <div class="q-cell bg-slate-700">
                        <div class="q-val top-0 left-0 w-full text-center">${cell.up.toFixed(1)}</div>
                        <div class="q-val bottom-0 left-0 w-full text-center">${cell.down.toFixed(1)}</div>
                        <div class="q-val top-2 left-0 w-full">${cell.left.toFixed(1)}</div>
                        <div class="q-val top-2 right-0 w-full">${cell.right.toFixed(1)}</div>
                    </div>
                `;
            });
        }

        function trainStep() {
            episodes++;
            let state = Math.floor(Math.random() * 9);
            let action = ['up', 'down', 'left', 'right'][Math.floor(Math.random() * 4)];
            qTable[state][action] += 0.1;
            document.getElementById('status').innerText = `Episodes: ${episodes}`;
            document.getElementById('terminal').innerHTML += `> State ${state}, Action ${action}: Q updated to ${qTable[state][action].toFixed(2)}<br>`;
            renderTable();
        }

        function askAI() {
            const q = document.getElementById('aiInput').value;
            document.getElementById('chatBox').innerHTML += `<p>> ${q}</p><p class="text-indigo-400">> Q-Learning balances exploration (trying new things) and exploitation (using known good actions) using the epsilon-greedy strategy.</p>`;
            document.getElementById('aiInput').value = '';
        }

        function resetSim() { qTable = Array(9).fill().map(() => ({up:0, down:0, left:0, right:0})); episodes=0; renderTable(); }
        renderTable();
    </script>
