    <script>
        let nodes = [
            { id: 'Root', type: 'max', v: null },
            { id: 'A', type: 'min', v: null }, { id: 'B', type: 'min', v: null },
            { id: 'L1', type: 'leaf', v: 3 }, { id: 'L2', type: 'leaf', v: 5 },
            { id: 'L3', type: 'leaf', v: 2 }, { id: 'L4', type: 'leaf', v: 9 }
        ];
        let step = 0;

        function switchTab(t) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('text-indigo-400', 'bg-slate-700/60', 'border', 'border-slate-600'));
            document.getElementById('content-' + t).classList.remove('hidden');
            const activeBtn = Array.from(document.querySelectorAll('.tab-btn')).find(b => b.innerText.toLowerCase().includes(t));
            if (activeBtn) activeBtn.classList.add('text-indigo-400', 'bg-slate-700/60', 'border', 'border-slate-600');
        }

        function renderTree() {
            const tree = document.getElementById('tree');
            tree.innerHTML = nodes.map(n => `
                <div class="node ${n.type}-node ${n.v !== null ? 'leaf-node' : ''}">
                    <span>${n.id}</span>
                    <span>${n.v !== null ? n.v : '?'}</span>
                </div>
            `).join('');
        }

        function runMinimax() {
            if (step === 0) {
                document.getElementById('terminal').innerHTML += "> Evaluating Leaf nodes...<br>";
                step++;
            } else if (step === 1) {
                nodes[1].v = 3; nodes[2].v = 2;
                document.getElementById('terminal').innerHTML += "> MIN nodes updated with min(leafs).<br>";
                step++;
            } else {
                nodes[0].v = 3;
                document.getElementById('terminal').innerHTML += "> MAX node updated with max(children). Done.<br>";
            }
            renderTree();
        }

        function askAI() {
            const q = document.getElementById('aiInput').value;
            document.getElementById('chatBox').innerHTML += `<p>> ${q}</p><p class="text-indigo-400">> Minimax assumes both players play optimally. One player maximizes their utility while the other minimizes it.</p>`;
            document.getElementById('aiInput').value = '';
        }
        
        renderTree();
