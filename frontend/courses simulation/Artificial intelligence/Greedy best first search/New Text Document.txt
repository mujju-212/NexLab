    <script>
        let nodes = [
            { id: 'A', h: 6 }, { id: 'B', h: 4 }, { id: 'C', h: 2 }, { id: 'D', h: 1 }, { id: 'G', h: 0 }
        ];
        let visited = [];
        let active = null;
        
        function switchTab(t) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('text-indigo-400', 'bg-slate-700/60', 'border', 'border-slate-600'));
            document.getElementById('content-' + t).classList.remove('hidden');
            const activeBtn = Array.from(document.querySelectorAll('.tab-btn')).find(b => b.innerText.toLowerCase().includes(t));
            if (activeBtn) activeBtn.classList.add('text-indigo-400', 'bg-slate-700/60', 'border', 'border-slate-600');
        }

        function renderGraph() {
            const graph = document.getElementById('graph');
            graph.innerHTML = nodes.map(n => `
                <div class="node ${visited.includes(n.id) ? 'visited' : ''} ${active === n.id ? 'active-node' : ''}">
                    <span>${n.id}</span>
                    <span class="text-amber-400">h=${n.h}</span>
                </div>
            `).join('');
        }

        function runGBFS() {
            // Greedy: pick node with lowest h
            let candidates = nodes.filter(n => !visited.includes(n.id));
            if (candidates.length > 0) {
                let next = candidates.reduce((prev, curr) => prev.h < curr.h ? prev : curr);
                active = next.id;
                visited.push(active);
                document.getElementById('terminal').innerHTML += `> Expanding node ${next.id} (h=${next.h})...<br>`;
                document.getElementById('status').innerText = `Visited: [${visited.join(', ')}]`;
                renderGraph();
            }
        }

        function askAI() {
            const q = document.getElementById('aiInput').value;
            document.getElementById('chatBox').innerHTML += `<p>> ${q}</p><p class="text-indigo-400">> Heuristics are 'rules of thumb'. GBFS uses $h(n)$ to guess the cost to the goal, but it can be misled by local optima!</p>`;
            document.getElementById('aiInput').value = '';
        }
        
        renderGraph();
    </script>
