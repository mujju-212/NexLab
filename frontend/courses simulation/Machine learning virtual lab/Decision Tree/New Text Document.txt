        <div id="content-theory" class="tab-content bg-slate-800 border border-slate-700 rounded-xl p-6 shadow-md space-y-6">
            <h2 class="text-xl font-bold text-indigo-400 border-b border-slate-700 pb-2">Theory: Decision Tree Classification & Impurity Metrics</h2>
            <p class="text-sm text-slate-300 leading-relaxed">
                A <strong>Decision Tree</strong> is a non-parametric supervised learning algorithm used for classification and regression tasks. It recursively splits a dataset into smaller subsets based on feature values, forming a flowchart-like tree structure where internal nodes represent feature decision rules, branches represent test outcomes, and leaf nodes represent target class labels.
            </p>

            <div class="grid grid-cols-1 md:grid-cols-3 gap-4 text-sm">
                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-chart-pie text-emerald-400"></i> 1. Shannon Entropy $H(S)$</span>
                        <span class="text-[10px] bg-emerald-500/20 text-emerald-300 px-2 py-0.5 rounded font-sans">Impurity</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Measures randomness or disorder in a dataset:
                        <br><code class="text-amber-300 font-mono">H(S) = - \sum_{i=1}^{k} p_i \log_2(p_i)</code>.
                        <br>For binary class with equal split ($50/50$), $H(S) = 1.0$ (Max Impurity). For pure set, $H(S) = 0.0$.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-filter text-amber-400"></i> 2. Information Gain ($IG$)</span>
                        <span class="text-[10px] bg-amber-500/20 text-amber-300 px-2 py-0.5 rounded font-sans">Split Selection</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Measures the reduction in entropy achieved by partitioning data on feature $A$:
                        <br><code class="text-emerald-300 font-mono">IG(S, A) = H(S) - \sum \frac{|S_v|}{|S|} H(S_v)</code>.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-scale-balanced text-cyan-400"></i> 3. Gini Impurity $G(S)$</span>
                        <span class="text-[10px] bg-cyan-500/20 text-cyan-300 px-2 py-0.5 rounded font-sans">CART Criterion</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Probability of misclassifying a randomly chosen element:
                        <br><code class="text-amber-300 font-mono">G(S) = 1 - \sum_{i=1}^{k} p_i^2</code>.
                        <br>Faster to compute than entropy because it avoids logarithmic operations.
                    </p>
                </div>
            </div>

            <div class="bg-slate-900/60 p-4 rounded-lg border border-slate-700 text-xs text-slate-300 space-y-2">
                <h4 class="font-bold text-indigo-300 text-sm"><i class="fa-solid fa-circle-info text-amber-300"></i> Overfitting & Tree Pruning</h4>
                <p class="text-slate-400 leading-relaxed">
                    Unconstrained decision trees can grow deep enough to fit every single training sample, resulting in <strong>overfitting</strong> ($100\%$ training accuracy but poor generalization). Hyperparameters like <code class="text-emerald-300 font-mono">max_depth</code>, <code class="text-emerald-300 font-mono">min_samples_split</code>, and cost-complexity pruning limit tree depth to ensure robust performance.
                </p>
            </div>
        </div>

        function executeStepAtIndex(idx) {
            const step = stepList[idx];
            highlightLine(step.line);

            renderTreeStructure(step.activeLevel);

            appendConsole(step.consoleMsg);
            document.getElementById('explanationText').innerText = step.explanation;
            document.getElementById('currentStepBadge').innerText = `Epoch ${idx + 1}/${stepList.length}`;

            if (idx === stepList.length - 1) {
                showResultBanner("Decision Tree Model Fitted Successfully", false);
                const btnStep = document.getElementById('btnStep');
                btnStep.disabled = true;
                btnStep.className = "bg-slate-700 text-slate-400 font-medium px-4 py-2 rounded-lg text-sm transition cursor-not-allowed";
                document.getElementById('currentStepBadge').innerText = "Converged";
            }
        }

        function stepExecute() {
            if (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function runAll() {
            startSimulation();
            while (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function sendAiQuery() {
            const input = document.getElementById('aiQueryInput');
            const query = input.value.trim();
            if (!query) return;

            const chatBox = document.getElementById('aiChatBox');

            const userDiv = document.createElement('div');
            userDiv.className = 'flex items-start gap-3 justify-end';
            userDiv.innerHTML = `
                <div class="bg-indigo-600/30 border border-indigo-500/40 p-3 rounded-lg text-slate-200">
                    ${query}
                </div>
                <div class="w-7 h-7 rounded-full bg-slate-700 flex items-center justify-center text-slate-300 text-xs shrink-0">You</div>
            `;
            chatBox.appendChild(userDiv);
            input.value = '';

            let response = "Try testing Gini vs Entropy criteria or changing Max Depth in the simulator to see how node splits update!";
            const qLower = query.toLowerCase();
            if (qLower.includes('entropy') || qLower.includes('gini')) {
                response = "Entropy uses logarithms ($H = -\\sum p_i \\log_2 p_i$) whereas Gini Impurity uses squares ($G = 1 - \\sum p_i^2$). Both measure pure vs mixed nodes, but Gini is computationally faster!";
            } else if (qLower.includes('gain') || qLower.includes('split') || qLower.includes('feature')) {
                response = "Information Gain measures how much Entropy decreases after splitting on a feature: $IG = H(Parent) - H(Children)$. The tree chooses the feature with the highest Information Gain!";
            } else if (qLower.includes('overfit') || qLower.includes('depth') || qLower.includes('prun')) {
                response = "Deep decision trees overfit by creating rules for individual noise samples. Setting `max_depth=2` or `max_depth=3` prunes the tree to ensure strong generalization on unseen test data!";
            }

            setTimeout(() => {
                const aiDiv = document.createElement('div');
                aiDiv.className = 'flex items-start gap-3';
                aiDiv.innerHTML = `
                    <div class="w-7 h-7 rounded-full bg-indigo-600 flex items-center justify-center text-white text-xs shrink-0">AI</div>
                    <div class="bg-slate-800 p-3 rounded-lg border border-slate-700 text-slate-200">
                        ${response}
                    </div>
                `;
                chatBox.appendChild(aiDiv);
                chatBox.scrollTop = chatBox.scrollHeight;
            }, 500);
        }
