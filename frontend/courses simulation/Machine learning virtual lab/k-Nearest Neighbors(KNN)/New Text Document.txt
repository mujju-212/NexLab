        <!-- TAB 1: THEORY -->
        <div id="content-theory" class="tab-content bg-slate-800 border border-slate-700 rounded-xl p-6 shadow-md space-y-6">
            <h2 class="text-xl font-bold text-indigo-400 border-b border-slate-700 pb-2">Theory: k-Nearest Neighbors (KNN) & Distance Metrics</h2>
            <p class="text-sm text-slate-300 leading-relaxed">
                <strong>k-Nearest Neighbors (KNN)</strong> is a simple, non-parametric, instance-based (lazy learning) algorithm used for both classification and regression. Unlike parametric models (e.g., Logistic Regression), KNN does not learn explicit model parameters ($w, b$) during a training phase; instead, it stores the entire training dataset in memory and calculates distances to classify new query points at prediction time.
            </p>

            <div class="bg-slate-950 p-4 rounded-lg border border-slate-700 font-mono text-center text-indigo-300 text-sm space-y-1">
                <p>Euclidean Distance: $d(x, y) = \sqrt{\sum_{i=1}^{n} (x_i - y_i)^2}$</p>
                <p class="text-xs text-amber-300">Manhattan Distance: $d(x, y) = \sum_{i=1}^{n} |x_i - y_i|$</p>
            </div>

            <div class="grid grid-cols-1 md:grid-cols-3 gap-4 text-sm">
                
                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-hashtag text-emerald-400"></i> 1. Choice of $k$ Parameter</span>
                        <span class="text-[10px] bg-emerald-500/20 text-emerald-300 px-2 py-0.5 rounded font-sans">Hyperparameter</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        A small $k$ (e.g., $k=1$) is sensitive to noise/outliers (overfitting). A large $k$ creates smooth decision boundaries but can include points from neighboring classes (underfitting). Odd values prevent tie votes!
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-ruler-combined text-amber-400"></i> 2. Distance Metrics</span>
                        <span class="text-[10px] bg-amber-500/20 text-amber-300 px-2 py-0.5 rounded font-sans">Distance</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        <strong>Euclidean</strong> (L2 Norm) measures straight-line distance. <strong>Manhattan</strong> (L1 Norm / City Block) measures grid-like distance along axes. Feature scaling is essential!
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-check-to-slot text-cyan-400"></i> 3. Majority Voting</span>
                        <span class="text-[10px] bg-cyan-500/20 text-cyan-300 px-2 py-0.5 rounded font-sans">Decision Rule</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        After identifying the $k$ nearest training samples to a query point, KNN counts class occurrences. The query point is assigned to the class with the majority vote count.
                    </p>
                </div>

            </div>

            <div class="bg-slate-900/60 p-4 rounded-lg border border-slate-700 text-xs text-slate-300 space-y-2">
                <h4 class="font-bold text-indigo-300 text-sm"><i class="fa-solid fa-circle-info text-amber-300"></i> The Curse of Dimensionality</h4>
                <p class="text-slate-400">
                    As the number of features (dimensions) increases, the volume of feature space grows exponentially, making all data points appear equidistant. Feature selection and dimensionality reduction (like PCA) help maintain KNN effectiveness.
                </p>
            </div>
        </div>

        // Calculate Distances and KNN Neighbors
        function computeKNN(points, query, k, metric) {
            const calculated = points.map((p, idx) => {
                let d = 0;
                if (metric === 'euclidean') {
                    d = Math.sqrt(Math.pow(p.x - query.x, 2) + Math.pow(p.y - query.y, 2));
                } else { // Manhattan
                    d = Math.abs(p.x - query.x) + Math.abs(p.y - query.y);
                }
                return { ...p, dist: d, idx };
            });

            // Sort by distance ascending
            calculated.sort((a, b) => a.dist - b.dist);

            const neighbors = calculated.slice(0, k);
            let class0Count = 0;
            let class1Count = 0;

            neighbors.forEach(n => {
                if (n.label === 0) class0Count++;
                else class1Count++;
            });

            const predictedLabel = class1Count > class0Count ? 1 : 0;
            const maxDist = neighbors[neighbors.length - 1]?.dist || 0;

            return { neighbors, class0Count, class1Count, predictedLabel, maxDist };
        }

        function drawKNNPlot(points, query, k, showNeighbors = false) {
            const canvas = document.getElementById('knnCanvas');
            if (!canvas) return;
            const ctx = canvas.getContext('2d');
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            const padding = 45;
            const width = canvas.width - padding * 2;
            const height = canvas.height - padding * 2;

            const xVals = points.map(p => p.x).concat(query.x);
            const yVals = points.map(p => p.y).concat(query.y);

            const minX = Math.min(...xVals) * 0.8;
            const maxX = Math.max(...xVals) * 1.1;
            const minY = Math.min(...yVals) * 0.8;
            const maxY = Math.max(...yVals) * 1.1;

            const mapX = x => padding + ((x - minX) / (maxX - minX)) * width;
            const mapY = y => canvas.height - padding - ((y - minY) / (maxY - minY)) * height;

            const metric = document.getElementById('metricSelect').value;
            const knnResult = computeKNN(points, query, k, metric);

            // Draw Axes
            ctx.strokeStyle = "#475569";
            ctx.lineWidth = 1;
            ctx.beginPath();
            ctx.moveTo(padding, padding);
            ctx.lineTo(padding, canvas.height - padding);
            ctx.lineTo(canvas.width - padding, canvas.height - padding);
            ctx.stroke();

            // Draw Radius Bounding Circle if neighbors active
            if (showNeighbors && knnResult.neighbors.length > 0) {
                const kThNeighbor = knnResult.neighbors[knnResult.neighbors.length - 1];
                const radiusPx = Math.abs(mapX(query.x + knnResult.maxDist) - mapX(query.x));

                ctx.strokeStyle = "rgba(99, 102, 241, 0.4)";
                ctx.fillStyle = "rgba(99, 102, 241, 0.08)";
                ctx.lineWidth = 2;
                ctx.setLineDash([4, 4]);
                ctx.beginPath();
                ctx.arc(mapX(query.x), mapY(query.y), radiusPx, 0, Math.PI * 2);
                ctx.fill();
                ctx.stroke();
                ctx.setLineDash([]);

                // Draw vectors to neighbors
                knnResult.neighbors.forEach(n => {
                    ctx.strokeStyle = n.label === 1 ? "#10b981" : "#3b82f6";
                    ctx.lineWidth = 1.8;
                    ctx.beginPath();
                    ctx.moveTo(mapX(query.x), mapY(query.y));
                    ctx.lineTo(mapX(n.x), mapY(n.y));
                    ctx.stroke();
                });
            }

            // Draw Data Points
            points.forEach(p => {
                const isNeighbor = showNeighbors && knnResult.neighbors.some(n => n.idx === p.idx);

                ctx.fillStyle = p.label === 1 ? "#10b981" : "#3b82f6";
                ctx.beginPath();
                ctx.arc(mapX(p.x), mapY(p.y), isNeighbor ? 8 : 5, 0, Math.PI * 2);
                ctx.fill();

                ctx.strokeStyle = isNeighbor ? "#f59e0b" : "#ffffff";
                ctx.lineWidth = isNeighbor ? 2.5 : 1;
                ctx.stroke();
            });

            // Draw Query Test Point (Orange Star)
            ctx.fillStyle = "#f59e0b";
            ctx.strokeStyle = "#ffffff";
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.arc(mapX(query.x), mapY(query.y), 9, 0, Math.PI * 2);
            ctx.fill();
            ctx.stroke();

            // Label Query Star Point
            ctx.fillStyle = "#ffffff";
            ctx.font = "bold 10px monospace";
            ctx.fillText("Q", mapX(query.x) - 3, mapY(query.y) + 3);
        }

            } else {

            }

            const q2Box = document.getElementById('postq2-box');

            if (q2 === 'a') {
                score++;

            const input = document.getElementById('aiQueryInput');
            const query = input.value.trim();
            if (!query) return;

            const chatBox = document.getElementById('aiChatBox');

            const userDiv = document.createElement('div');
            userDiv.className = 'flex items-start gap-3 justify-end';
            userDiv.innerHTML = `
                <div class="bg-indigo-600/30 border border-indigo-500/40 p-3 rounded-lg text-slate-200">
                    ${query}
                </div>
                <div class="w-7 h-7 rounded-full bg-slate-700 flex items-center justify-center text-slate-300 text-xs shrink-0">You</div>
            `;
            chatBox.appendChild(userDiv);
            input.value = '';

            let response = "Try clicking on the plot canvas to move the Query Point and observe how distance lines and majority votes update dynamically!";
            const qLower = query.toLowerCase();
            if (qLower.includes('euclidean') || qLower.includes('manhattan') || qLower.includes('distance')) {
                response = "Euclidean distance measures straight-line distance ($d = \\sqrt{\\Delta x^2 + \\spacing \\Delta y^2}$), whereas Manhattan distance measures grid movement along perpendicular axes ($d = |\\Delta x| + |\\Delta y|$).";
            } else if (qLower.includes('k value') || qLower.includes('choice of k') || qLower.includes('overfit')) {
                response = "Small $k$ (e.g. $k=1$) fits noise and causes overfitting. Large $k$ smoothes decision boundaries but risks underfitting. Odd values prevent tie votes in binary classification!";
            } else if (qLower.includes('curse') || qLower.includes('dimension')) {
                response = "In high dimensions, points become sparse and almost equidistant, degrading distance-based metrics like KNN. Feature scaling or PCA dimensionality reduction is essential!";
            }

            setTimeout(() => {
                const aiDiv = document.createElement('div');
                aiDiv.className = 'flex items-start gap-3';
                aiDiv.innerHTML = `
                    <div class="w-7 h-7 rounded-full bg-indigo-600 flex items-center justify-center text-white text-xs shrink-0">AI</div>
                    <div class="bg-slate-800 p-3 rounded-lg border border-slate-700 text-slate-200">
                        ${response}
                    </div>
                `;
                chatBox.appendChild(aiDiv);
                chatBox.scrollTop = chatBox.scrollHeight;
            }, 500);
        }
