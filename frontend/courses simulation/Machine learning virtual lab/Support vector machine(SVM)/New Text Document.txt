        <div id="content-theory" class="tab-content bg-slate-800 border border-slate-700 rounded-xl p-6 shadow-md space-y-6">
            <h2 class="text-xl font-bold text-indigo-400 border-b border-slate-700 pb-2">Theory: Support Vector Machine (SVM) Classification</h2>
            <p class="text-sm text-slate-300 leading-relaxed">
                A <strong>Support Vector Machine (SVM)</strong> is a powerful supervised learning algorithm used for binary classification and regression tasks. SVM finds an optimal decision boundary (a <strong>hyperplane</strong>) that maximizes the geometric margin between data points of two distinct classes $y_i \in \{-1, +1\}$.
            </p>

            <div class="grid grid-cols-1 md:grid-cols-3 gap-4 text-sm">
                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-arrows-left-right-to-line text-emerald-400"></i> 1. Maximum Margin Hyperplane</span>
                        <span class="text-[10px] bg-emerald-500/20 text-emerald-300 px-2 py-0.5 rounded font-sans">Optimization</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        SVM maximizes the margin width $\frac{2}{\|w\|}$ separating class boundaries:
                        <br><code class="text-amber-300 font-mono">\min_{w,b} \frac{1}{2}\|w\|^2 \quad \text{s.t. } y_i(w \cdot x_i + b) \ge 1</code>.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-circle-dot text-amber-400"></i> 2. Support Vectors</span>
                        <span class="text-[10px] bg-amber-500/20 text-amber-300 px-2 py-0.5 rounded font-sans">Critical Samples</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        The subset of training points lying exactly on or inside the margin boundaries $y_i(w \cdot x_i + b) = 1$. They fully define the position and orientation of the decision boundary!
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-sliders text-cyan-400"></i> 3. Penalty Parameter $C$</span>
                        <span class="text-[10px] bg-cyan-500/20 text-cyan-300 px-2 py-0.5 rounded font-sans">Soft Margin</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Controls trade-off between maximizing margin width and minimizing misclassifications. Small $C$ yields a wider margin with more allowed errors (high regularization), while large $C$ strictly classifies all points (risk of overfitting).
                    </p>
                </div>
            </div>

            <div class="bg-slate-900/60 p-4 rounded-lg border border-slate-700 text-xs text-slate-300 space-y-2">
                <h4 class="font-bold text-indigo-300 text-sm"><i class="fa-solid fa-circle-info text-amber-300"></i> The Kernel Trick for Non-Linear Data</h4>
                <p class="text-slate-400 leading-relaxed">
                    When data is not linearly separable in the original input space, the <strong>Kernel Trick</strong> implicitly maps input feature vectors into a higher-dimensional feature space where linear separation is possible without computing high-dimensional coordinates explicitly:
                </p>
                <ul class="list-disc list-inside space-y-1 text-slate-400 font-mono">
                    <li><strong class="text-slate-200 font-sans">Linear Kernel:</strong> $K(x, z) = x^T z$ (for linearly separable features)</li>
                    <li><strong class="text-slate-200 font-sans">RBF Kernel (Gaussian):</strong> $K(x, z) = \exp(-\gamma \|x - z\|^2)$ (for complex circular/curved boundaries)</li>
                    <li><strong class="text-slate-200 font-sans">Polynomial Kernel:</strong> $K(x, z) = (x^T z + r)^d$ (for polynomial decision surfaces)</li>
                </ul>
            </div>
        </div>

        function drawCanvasPlot(points, kernel, cVal, isFitted = false) {
            const canvas = document.getElementById('svmCanvas');
            if (!canvas) return;
            const ctx = canvas.getContext('2d');
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            const padding = 35;
            const width = canvas.width - padding * 2;
            const height = canvas.height - padding * 2;

            const mapX = x => padding + (x / 10) * width;
            const mapY = y => canvas.height - padding - (y / 8) * height;

            // Draw Background Grid
            ctx.strokeStyle = "#1e293b";
            ctx.lineWidth = 1;
            for (let x = 0; x <= 10; x += 2) {
                ctx.beginPath();
                ctx.moveTo(mapX(x), padding);
                ctx.lineTo(mapX(x), canvas.height - padding);
                ctx.stroke();
            }
            for (let y = 0; y <= 8; y += 2) {
                ctx.beginPath();
                ctx.moveTo(padding, mapY(y));
                ctx.lineTo(canvas.width - padding, mapY(y));
                ctx.stroke();
            }

            // Draw Decision Boundary & Margins when fitted
            if (isFitted) {
                if (kernel === 'linear') {
                    // Linear Hyperplane: y = -x + 9
                    ctx.strokeStyle = "#6366f1"; // Solid Hyperplane
                    ctx.lineWidth = 3;
                    ctx.beginPath();
                    ctx.moveTo(mapX(2.2), mapY(6.8));
                    ctx.lineTo(mapX(6.8), mapY(2.2));
                    ctx.stroke();

                    // Dashed Margin Lines
                    ctx.strokeStyle = "#a5b4fc";
                    ctx.setLineDash([5, 5]);
                    ctx.lineWidth = 1.5;

                    ctx.beginPath(); // Upper Margin (+1)
                    ctx.moveTo(mapX(3.0), mapY(7.2));
                    ctx.lineTo(mapX(7.6), mapY(2.6));
                    ctx.stroke();

                    ctx.beginPath(); // Lower Margin (-1)
                    ctx.moveTo(mapX(1.4), mapY(6.4));
                    ctx.lineTo(mapX(6.0), mapY(1.8));
                    ctx.stroke();

                    ctx.setLineDash([]);
                } else if (kernel === 'rbf') {
                    // Non-linear circular decision contour
                    ctx.strokeStyle = "#6366f1";
                    ctx.lineWidth = 3;
                    ctx.beginPath();
                    ctx.arc(mapX(5.0), mapY(5.0), width * 0.18, 0, Math.PI * 2);
                    ctx.stroke();

                    ctx.strokeStyle = "#a5b4fc";
                    ctx.setLineDash([4, 4]);
                    ctx.beginPath();
                    ctx.arc(mapX(5.0), mapY(5.0), width * 0.25, 0, Math.PI * 2);
                    ctx.stroke();
                    ctx.setLineDash([]);
                } else if (kernel === 'poly') {
                    // Non-linear polynomial curve
                    ctx.strokeStyle = "#6366f1";
                    ctx.lineWidth = 3;
                    ctx.beginPath();
                    ctx.moveTo(mapX(1.5), mapY(2.5));
                    ctx.bezierCurveTo(mapX(4.0), mapY(6.5), mapX(6.0), mapY(2.0), mapX(8.5), mapY(5.5));
                    ctx.stroke();
                }
            }

            // Identify Support Vectors indices based on dataset
            let svIndices = [1, 3, 6, 8];
            if (scenario === 'circles') svIndices = [0, 2, 6, 7, 8];
            if (scenario === 'moons') svIndices = [2, 3, 5, 6, 9];

            points.forEach((p, idx) => {
                const px = mapX(p.x);
                const py = mapY(p.y);

                const isSv = isFitted && svIndices.includes(idx);

                // Draw SV Highlight Outer Ring
                if (isSv) {
                    ctx.strokeStyle = "#f59e0b"; // Glowing Gold Ring
                    ctx.lineWidth = 2.5;
                    ctx.beginPath();
                    ctx.arc(px, py, 10, 0, Math.PI * 2);
                    ctx.stroke();
                }

                // Draw Data Point Circle
                ctx.fillStyle = p.label === 1 ? "#10b981" : "#3b82f6";
                ctx.beginPath();
                ctx.arc(px, py, 6, 0, Math.PI * 2);
                ctx.fill();
                ctx.strokeStyle = "#ffffff";
                ctx.lineWidth = 1.5;
                ctx.stroke();
            });
        }

            document.getElementById('valAccuracy').innerText = `${step.accuracy}%`;

            document.getElementById('metricsBadge').innerText = `SVs: ${step.svCount} | Margin: ${step.marginW}`;

            drawCanvasPlot(datasets[scenario].points, kernel, cVal, step.isFitted);

            appendConsole(step.consoleMsg);
            document.getElementById('explanationText').innerText = step.explanation;
            document.getElementById('currentStepBadge').innerText = `Epoch ${idx + 1}/${stepList.length}`;

                showResultBanner("SVM Decision Hyperplane & Margins Optimized", false);
                const btnStep = document.getElementById('btnStep');
                btnStep.disabled = true;
                btnStep.className = "bg-slate-700 text-slate-400 font-medium px-4 py-2 rounded-lg text-sm transition cursor-not-allowed";
                document.getElementById('currentStepBadge').innerText = "Converged";
            }
        }

        function stepExecute() {

                aiDiv.className = 'flex items-start gap-3';

            }, 500);
