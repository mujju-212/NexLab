        <div id="content-theory" class="tab-content bg-slate-800 border border-slate-700 rounded-xl p-6 shadow-md space-y-6">
            <h2 class="text-xl font-bold text-indigo-400 border-b border-slate-700 pb-2">Theory: Dimensionality Reduction (PCA)</h2>
            <p class="text-sm text-slate-300 leading-relaxed">
                <strong>Principal Component Analysis (PCA)</strong> is a linear dimensionality reduction technique that transforms a dataset with many correlated features into a smaller set of uncorrelated features called <strong>Principal Components</strong>. It does this by identifying the directions (eigenvectors) of maximum variance in the data.
            </p>

            <div class="grid grid-cols-1 md:grid-cols-3 gap-4 text-sm">
                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-up-down-left-right text-emerald-400"></i> 1. Covariance Matrix</span>
                        <span class="text-[10px] bg-emerald-500/20 text-emerald-300 px-2 py-0.5 rounded font-sans">Relationships</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Measures how variables change together. PCA computes eigenvectors of this matrix to find the orthogonal axes of highest spread.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-chart-line text-amber-400"></i> 2. Explained Variance</span>
                        <span class="text-[10px] bg-amber-500/20 text-amber-300 px-2 py-0.5 rounded font-sans">Scree Plot</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Each PC captures a portion of the total variance. <code class="text-amber-300">explained_variance_ratio_</code> ranks these components by importance.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-compress text-cyan-400"></i> 3. Data Projection</span>
                        <span class="text-[10px] bg-cyan-500/20 text-cyan-300 px-2 py-0.5 rounded font-sans">Mapping</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Projects high-dimensional data onto the selected top $k$ principal components, achieving dimensionality compression while retaining the essential variance structure.
                    </p>
                </div>
            </div>
        </div>

    <script>
        let step = 0;
        const code = [
            "from sklearn.decomposition import PCA",
            "pca = PCA(n_components=2)",
            "X_projected = pca.fit_transform(X_high_dim)",
            "var_ratio = pca.explained_variance_ratio_",
            "print(f'Explained Variance: {var_ratio}')"
        ];

        function switchTab(tab) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.getElementById('content-' + tab).classList.remove('hidden');
        }

        function highlightLine(line) {
            document.getElementById('codeTrace').innerHTML = code.map((l, i) => 
                `<div class="code-line ${i === line ? 'active' : ''}">${l}</div>`
            ).join('');
        }

        function startSimulation() {
            step = 0;
            document.getElementById('btnStep').disabled = false;
            document.getElementById('btnStep').className = "bg-emerald-600 px-4 py-2 rounded-lg text-white font-medium text-sm";
            highlightLine(0);
            document.getElementById('terminal').innerHTML = ">>> Initializing PCA pipeline...";
            document.getElementById('explanationText').innerText = "Step 1: Importing PCA from sklearn.";
        }

        function stepExecute() {
            step++;
            if (step >= code.length) {
                document.getElementById('btnStep').disabled = true;
                return;
            }
            highlightLine(step);
            document.getElementById('terminal').innerHTML += `<br>>>> Line ${step} executed.`;
            document.getElementById('explanationText').innerText = "Processing step: " + code[step];
        }

        function checkPretest() {
            const val = document.querySelector('input[name="pq1"]:checked')?.value;
            if (val === 'b') {
                document.getElementById('preq1-box').classList.add('border-emerald-500');
                alert("Correct!");
            } else {
                document.getElementById('preq1-box').classList.add('border-rose-500');
            }
        }
    </script>
