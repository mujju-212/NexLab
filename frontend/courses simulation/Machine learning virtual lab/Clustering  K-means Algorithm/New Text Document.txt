        <!-- TAB 1: THEORY -->
        <div id="content-theory" class="tab-content bg-slate-800 border border-slate-700 rounded-xl p-6 shadow-md space-y-6">
            <h2 class="text-xl font-bold text-indigo-400 border-b border-slate-700 pb-2">Theory: Unsupervised K-Means Clustering Algorithm</h2>
            <p class="text-sm text-slate-300 leading-relaxed">
                <strong>K-Means</strong> is an unsupervised vector quantization and clustering algorithm that partitions $N$ unlabeled data points into $K$ distinct, non-overlapping clusters. Each cluster is represented by its geometric center called a <strong>Centroid</strong> ($\mu_k$).
            </p>

            <div class="grid grid-cols-1 md:grid-cols-3 gap-4 text-sm">
                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-bullseye text-emerald-400"></i> 1. Objective / Inertia $J$</span>
                        <span class="text-[10px] bg-emerald-500/20 text-emerald-300 px-2 py-0.5 rounded font-sans">Cost Function</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Minimizes Within-Cluster Sum of Squares (WCSS / Inertia):
                        <br><code class="text-amber-300 font-mono">J = \sum_{k=1}^{K} \sum_{x \in S_k} \|x - \mu_k\|^2</code>.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-arrows-spin text-amber-400"></i> 2. Expectation-Maximization</span>
                        <span class="text-[10px] bg-amber-500/20 text-amber-300 px-2 py-0.5 rounded font-sans">Two Steps</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        <strong>E-Step (Assignment):</strong> Assign each sample to its nearest centroid $c^{(i)} = \arg\min_k \|x^{(i)} - \mu_k\|^2$.
                        <br><strong>M-Step (Update):</strong> Recalculate centroids $\mu_k = \frac{1}{|S_k|} \sum_{x \in S_k} x$.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-chart-line text-cyan-400"></i> 3. The Elbow Method</span>
                        <span class="text-[10px] bg-cyan-500/20 text-cyan-300 px-2 py-0.5 rounded font-sans">Choosing K</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Plotting Inertia vs. $K$ yields a steep decline followed by a bend (an "elbow"). The point where the rate of decrease abruptly flattens indicates the optimal number of clusters.
                    </p>
                </div>
            </div>

            <div class="bg-slate-900/60 p-4 rounded-lg border border-slate-700 text-xs text-slate-300 space-y-2">
                <h4 class="font-bold text-indigo-300 text-sm"><i class="fa-solid fa-circle-info text-amber-300"></i> Centroid Initialization: Random vs. K-Means++</h4>
                <p class="text-slate-400 leading-relaxed">
                    Standard random initialization can get trapped in poor local minima if initial centroids start close together. <strong>K-Means++</strong> initializes centroids smart-proportional to distance squared ($\propto d(x)^2$), spreading initial centers far apart for faster and reliable convergence!
                </p>
            </div>
        </div>

        function drawSpatialCanvas(points, centroids, history = [], isFitted = false) {
            const canvas = document.getElementById('kmeansCanvas');
            if (!canvas) return;
            const ctx = canvas.getContext('2d');
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            const padding = 35;
            const width = canvas.width - padding * 2;
            const height = canvas.height - padding * 2;

            const mapX = x => padding + (x / 10) * width;
            const mapY = y => canvas.height - padding - (y / 8) * height;

            // Background Grid
            ctx.strokeStyle = "#1e293b";
            ctx.lineWidth = 1;
            for (let x = 0; x <= 10; x += 2) {
                ctx.beginPath();
                ctx.moveTo(mapX(x), padding);
                ctx.lineTo(mapX(x), canvas.height - padding);
                ctx.stroke();
            }
            for (let y = 0; y <= 8; y += 2) {
                ctx.beginPath();
                ctx.moveTo(padding, mapY(y));
                ctx.lineTo(canvas.width - padding, mapY(y));
                ctx.stroke();
            }

            // Draw Centroid Trajectory Lines
            if (history.length > 1) {
                const k = centroids.length;
                for (let cIdx = 0; cIdx < k; cIdx++) {
                    ctx.strokeStyle = clusterColors[cIdx % clusterColors.length];
                    ctx.setLineDash([3, 3]);
                    ctx.lineWidth = 1.5;
                    ctx.beginPath();
                    for (let hIdx = 0; hIdx < history.length; hIdx++) {
                        const pt = history[hIdx][cIdx];
                        if (hIdx === 0) ctx.moveTo(mapX(pt.x), mapY(pt.y));
                        else ctx.lineTo(mapX(pt.x), mapY(pt.y));
                    }
                    ctx.stroke();
                    ctx.setLineDash([]);
                }
            }

            // Draw Data Points
            points.forEach(p => {
                const px = mapX(p.x);
                const py = mapY(p.y);
                const clusterIdx = isFitted && p.cluster !== undefined ? p.cluster : -1;

                ctx.fillStyle = clusterIdx >= 0 ? clusterColors[clusterIdx % clusterColors.length] : "#94a3b8";
                ctx.beginPath();
                ctx.arc(px, py, 5, 0, Math.PI * 2);
                ctx.fill();
                ctx.strokeStyle = "#ffffff";
                ctx.lineWidth = 1;
                ctx.stroke();
            });

            // Draw Centroid Markers (Glowing Large Stars/Crosshairs)
            centroids.forEach((c, idx) => {
                const cx = mapX(c.x);
                const cy = mapY(c.y);
                const color = clusterColors[idx % clusterColors.length];

                // Outer Halo
                ctx.strokeStyle = color;
                ctx.lineWidth = 3;
                ctx.beginPath();
                ctx.arc(cx, cy, 11, 0, Math.PI * 2);
                ctx.stroke();

                // Crosshair Plus
                ctx.strokeStyle = "#ffffff";
                ctx.lineWidth = 2;
                ctx.beginPath();
                ctx.moveTo(cx - 7, cy);
                ctx.lineTo(cx + 7, cy);
                ctx.moveTo(cx, cy - 7);
                ctx.lineTo(cx, cy + 7);
                ctx.stroke();
            });
        }

        function calculateKMeansIterations() {
            const k = parseInt(document.getElementById('kSelect').value) || 3;
            const initStrategy = document.getElementById('initSelect').value;
            let pts = JSON.parse(JSON.stringify(currentPoints));

            // Seed Initial Centroids
            let cents = [];
            if (initStrategy === 'kmeans++' || scenario === 'blobs3') {
                cents = [
                    { x: 2.0, y: 2.0 },
                    { x: 8.0, y: 7.0 },
                    { x: 8.0, y: 2.0 },
                    { x: 2.5, y: 7.5 },
                    { x: 5.0, y: 5.0 }
                ].slice(0, k);
            } else {
                cents = [
                    { x: 1.0, y: 1.0 },
                    { x: 3.0, y: 3.0 },
                    { x: 5.0, y: 5.0 },
                    { x: 7.0, y: 7.0 },
                    { x: 9.0, y: 9.0 }
                ].slice(0, k);
            }

            let history = [JSON.parse(JSON.stringify(cents))];
            let maxIter = 4;

            let iterationsData = [];

            for (let iter = 0; iter < maxIter; iter++) {
                // E-Step: Assign Points to Nearest Centroid
                let sse = 0;
                pts.forEach(p => {
                    let minD = Infinity;
                    let bestC = 0;
                    cents.forEach((c, cIdx) => {
                        let d = Math.pow(p.x - c.x, 2) + Math.pow(p.y - c.y, 2);
                        if (d < minD) {
                            minD = d;
                            bestC = cIdx;
                        }
                    });
                    p.cluster = bestC;
                    sse += minD;
                });

                // M-Step: Recalculate Centroid Coordinates
                let newCents = cents.map(() => ({ x: 0, y: 0, count: 0 }));
                pts.forEach(p => {
                    newCents[p.cluster].x += p.x;
                    newCents[p.cluster].y += p.y;
                    newCents[p.cluster].count += 1;
                });

                let totalShift = 0;
                cents = newCents.map((nc, idx) => {
                    if (nc.count > 0) {
                        let updatedX = parseFloat((nc.x / nc.count).toFixed(2));
                        let updatedY = parseFloat((nc.y / nc.count).toFixed(2));
                        let shift = Math.sqrt(Math.pow(updatedX - cents[idx].x, 2) + Math.pow(updatedY - cents[idx].y, 2));
                        totalShift += shift;
                        return { x: updatedX, y: updatedY };
                    }
                    return cents[idx];
                });

                history.push(JSON.parse(JSON.stringify(cents)));

                iterationsData.push({
                    iter: iter + 1,
                    pts: JSON.parse(JSON.stringify(pts)),
                    cents: JSON.parse(JSON.stringify(cents)),
                    history: JSON.parse(JSON.stringify(history)),
                    sse: parseFloat(sse.toFixed(2)),
                    shift: parseFloat(totalShift.toFixed(3))
                });

                if (totalShift < 0.01) break;
            }

            return iterationsData;
        }

        function sendAiQuery() {
            const input = document.getElementById('aiQueryInput');
            const query = input.value.trim();
            if (!query) return;

            const chatBox = document.getElementById('aiChatBox');

            const userDiv = document.createElement('div');
            userDiv.className = 'flex items-start gap-3 justify-end';
            userDiv.innerHTML = `
                <div class="bg-indigo-600/30 border border-indigo-500/40 p-3 rounded-lg text-slate-200">
                    ${query}
                </div>
                <div class="w-7 h-7 rounded-full bg-slate-700 flex items-center justify-center text-slate-300 text-xs shrink-0">You</div>
            `;
            chatBox.appendChild(userDiv);
            input.value = '';

            let response = "Try stepping through iterations or testing different values of K in the simulator to see how centroids shift and inertia decreases!";
            const qLower = query.toLowerCase();
            if (qLower.includes('elbow') || qLower.includes('k') || qLower.includes('inertia')) {
                response = "The Elbow Method plots Inertia (WCSS) against $K$. The 'elbow' point represents a threshold where adding more centroids yields diminishing returns in variance reduction!";
            } else if (qLower.includes('centroid') || qLower.includes('update') || qLower.includes('shift')) {
                response = "Centroids update during the M-step by taking the mean of all data points assigned to that cluster: $\\mu_k = \\frac{1}{|S_k|} \\sum_{x \\in S_k} x$. Once centroids stop shifting, the algorithm converges!";
            } else if (qLower.includes('kmeans++') || qLower.includes('init') || qLower.includes('random')) {
                response = "K-Means++ initializes centroids far apart by choosing subsequent centers proportional to squared distance $d(x)^2$ from existing centroids, preventing poor local minima!";
            }

            setTimeout(() => {
                const aiDiv = document.createElement('div');
                aiDiv.className = 'flex items-start gap-3';
                aiDiv.innerHTML = `
                    <div class="w-7 h-7 rounded-full bg-indigo-600 flex items-center justify-center text-white text-xs shrink-0">AI</div>
                    <div class="bg-slate-800 p-3 rounded-lg border border-slate-700 text-slate-200">
                        ${response}
                    </div>
                `;
                chatBox.appendChild(aiDiv);
                chatBox.scrollTop = chatBox.scrollHeight;
            }, 500);
        }
