        <!-- TAB 1: THEORY -->
        <div id="content-theory" class="tab-content bg-slate-800 border border-slate-700 rounded-xl p-6 shadow-md space-y-6">
            <h2 class="text-xl font-bold text-indigo-400 border-b border-slate-700 pb-2">Theory: Linear Regression & Gradient Descent Optimization</h2>
            <p class="text-sm text-slate-300 leading-relaxed">
                <strong>Linear Regression</strong> is a foundational supervised learning algorithm used to model the linear relationship between an independent input variable $X$ and a continuous dependent target variable $Y$. The hypothesis function is modeled as a line:
            </p>

            <div class="bg-slate-950 p-4 rounded-lg border border-slate-700 font-mono text-center text-indigo-300 text-sm">
                $\hat{y} = w \cdot x + b$ &nbsp;&nbsp;&nbsp;&nbsp; where $w$ = Slope / Weight, $b$ = Y-Intercept / Bias
            </div>

            <div class="grid grid-cols-1 md:grid-cols-3 gap-4 text-sm">
                
                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-bullseye text-emerald-400"></i> 1. Cost Function (MSE)</span>
                        <span class="text-[10px] bg-emerald-500/20 text-emerald-300 px-2 py-0.5 rounded font-sans">Loss Metric</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Measures average squared error between actual targets $y_i$ and predictions $\hat{y}_i$:
                        <br><code class="text-amber-300 font-mono">J(w, b) = (1 / 2n) * Σ (y_i - (w*x_i + b))^2</code>.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-stairs text-amber-400"></i> 2. Gradient Descent</span>
                        <span class="text-[10px] bg-amber-500/20 text-amber-300 px-2 py-0.5 rounded font-sans">Optimization</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Iteratively updates parameters $w$ and $b$ in the direction of steepest loss reduction:
                        <br><code class="text-emerald-300 font-mono">w = w - α * (∂J / ∂w)</code>, <code class="text-emerald-300 font-mono">b = b - α * (∂J / ∂b)</code>.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-gauge-high text-cyan-400"></i> 3. R² Coefficient of Determination</span>
                        <span class="text-[10px] bg-cyan-500/20 text-cyan-300 px-2 py-0.5 rounded font-sans">Metric [0 to 1]</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Evaluates model fit goodness ($0.0 \to 1.0$). $R^2 = 1 - \frac{\text{SS}_{res}}{\text{SS}_{tot}}$, representing the proportion of variance explained.
                    </p>
                </div>

            </div>

            <div class="bg-slate-900/60 p-4 rounded-lg border border-slate-700 text-xs text-slate-300 space-y-2">
                <h4 class="font-bold text-indigo-300 text-sm"><i class="fa-solid fa-circle-info text-amber-300"></i> Role of Learning Rate ($\alpha$)</h4>
                <p class="text-slate-400">
                    If learning rate $\alpha$ is too small, gradient descent converges extremely slowly. If $\alpha$ is too large, the updates overshoot the minimum cost and can diverge completely!
                </p>
            </div>
        </div>

        function drawPlot(points, w, b, drawLine = false) {
            const canvas = document.getElementById('regressionCanvas');
            if (!canvas) return;
            const ctx = canvas.getContext('2d');
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            const padding = 40;
            const width = canvas.width - padding * 2;
            const height = canvas.height - padding * 2;

            const xVals = points.map(p => p.x);
            const yVals = points.map(p => p.y);

            const minX = Math.min(...xVals) * 0.8;
            const maxX = Math.max(...xVals) * 1.1;
            const minY = Math.min(...yVals) * 0.8;
            const maxY = Math.max(...yVals) * 1.1;

            const mapX = x => padding + ((x - minX) / (maxX - minX)) * width;
            const mapY = y => canvas.height - padding - ((y - minY) / (maxY - minY)) * height;

            // Draw Axes
            ctx.strokeStyle = "#475569";
            ctx.lineWidth = 1;
            ctx.beginPath();
            ctx.moveTo(padding, padding);
            ctx.lineTo(padding, canvas.height - padding);
            ctx.lineTo(canvas.width - padding, canvas.height - padding);
            ctx.stroke();

            // Draw Residual Lines if regression line active
            if (drawLine) {
                ctx.strokeStyle = "rgba(244, 63, 94, 0.4)";
                ctx.lineWidth = 1.5;
                points.forEach(p => {
                    const yPred = w * p.x + b;
                    ctx.beginPath();
                    ctx.moveTo(mapX(p.x), mapY(p.y));
                    ctx.lineTo(mapX(p.x), mapY(yPred));
                    ctx.stroke();
                });

                // Draw Regression Line
                const x1 = minX;
                const y1 = w * x1 + b;
                const x2 = maxX;
                const y2 = w * x2 + b;

                ctx.strokeStyle = "#10b981";
                ctx.lineWidth = 3;
                ctx.beginPath();
                ctx.moveTo(mapX(x1), mapY(y1));
                ctx.lineTo(mapX(x2), mapY(y2));
                ctx.stroke();
            }

            // Draw Data Points
            points.forEach(p => {
                ctx.fillStyle = "#6366f1";
                ctx.beginPath();
                ctx.arc(mapX(p.x), mapY(p.y), 5, 0, Math.PI * 2);
                ctx.fill();
                ctx.strokeStyle = "#ffffff";
                ctx.lineWidth = 1;
                ctx.stroke();
            });
        }

        function calculateOLS(points) {
            const n = points.length;
            const sumX = points.reduce((s, p) => s + p.x, 0);
            const sumY = points.reduce((s, p) => s + p.y, 0);
            const sumXY = points.reduce((s, p) => s + p.x * p.y, 0);
            const sumX2 = points.reduce((s, p) => s + p.x * p.x, 0);

            const w = (n * sumXY - sumX * sumY) / (n * sumX2 - sumX * sumX);
            const b = (sumY - w * sumX) / n;

            const yMean = sumY / n;
            const ssTot = points.reduce((s, p) => s + Math.pow(p.y - yMean, 2), 0);
            const ssRes = points.reduce((s, p) => s + Math.pow(p.y - (w * p.x + b), 2), 0);
            const r2 = 1 - (ssRes / ssTot);
            const mse = ssRes / n;

            return { w, b, r2, mse };
        }

        function buildExecutionSteps() {
            const data = datasets[scenario].points;
            const ols = calculateOLS(data);

            const lr = parseFloat(document.getElementById('learningRate').value) || 0.01;

            stepList = [

                    line: 1,
                    w: 0.1 * ols.w,
                    b: 0.1 * ols.b,
                    mse: ols.mse * 4.5,
                    r2: 0.12,

                    explanation: `Line 1: Initialized regression model and dataset.`

        }

        function startSimulation() {
            updateDatasetDisplay();
            buildExecutionSteps();
            currentStepIdx = 0;

            appendConsole(`>>> Executing: ml_linear_regression.py [Dataset: ${scenario}]`);
            executeStepAtIndex(0);
        }

        function executeStepAtIndex(idx) {
            const step = stepList[idx];
            highlightLine(step.line);

            document.getElementById('valW').innerText = step.w.toFixed(3);
            document.getElementById('valB').innerText = step.b.toFixed(3);
            document.getElementById('valMSE').innerText = step.mse.toFixed(2);
            document.getElementById('metricsBadge').innerText = `R²: ${step.r2.toFixed(3)} | MSE: ${step.mse.toFixed(2)}`;

            drawPlot(datasets[scenario].points, step.w, step.b, true);

            appendConsole(step.consoleMsg);
            document.getElementById('explanationText').innerText = step.explanation;
            document.getElementById('currentStepBadge').innerText = `Epoch ${idx + 1}/${stepList.length}`;

            if (idx === stepList.length - 1) {
                showResultBanner("Linear Regression Convergence Completed", false);
                const btnStep = document.getElementById('btnStep');
                btnStep.disabled = true;
                btnStep.className = "bg-slate-700 text-slate-400 font-medium px-4 py-2 rounded-lg text-sm transition cursor-not-allowed";
                document.getElementById('currentStepBadge').innerText = "Converged";
            }
        }

        function stepExecute() {
            if (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function runAll() {
            startSimulation();
            while (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }
