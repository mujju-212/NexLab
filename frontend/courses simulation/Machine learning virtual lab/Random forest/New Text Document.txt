        <!-- TAB 1: THEORY -->
        <div id="content-theory" class="tab-content bg-slate-800 border border-slate-700 rounded-xl p-6 shadow-md space-y-6">
            <h2 class="text-xl font-bold text-indigo-400 border-b border-slate-700 pb-2">Theory: Random Forest Classification & Ensemble Learning</h2>
            <p class="text-sm text-slate-300 leading-relaxed">
                A <strong>Random Forest</strong> is an <strong>Ensemble Learning</strong> algorithm that combines multiple Decision Trees using <strong>Bagging (Bootstrap Aggregating)</strong> and the <strong>Random Subspace Method</strong>. By aggregating predictions from a diverse "forest" of weak base decision trees, it significantly reduces model variance, combats overfitting, and delivers superior generalization performance.
            </p>

            <div class="grid grid-cols-1 md:grid-cols-3 gap-4 text-sm">
                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-arrows-split-up-and-left text-emerald-400"></i> 1. Bootstrap Aggregating (Bagging)</span>
                        <span class="text-[10px] bg-emerald-500/20 text-emerald-300 px-2 py-0.5 rounded font-sans">Data Sampling</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Each tree is trained on a random sample drawn from the training set with replacement. Unselected samples form the <strong>Out-Of-Bag (OOB)</strong> set used for unbiased error estimation.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-random text-amber-400"></i> 2. Random Subspace Method</span>
                        <span class="text-[10px] bg-amber-500/20 text-amber-300 px-2 py-0.5 rounded font-sans">Feature Sampling</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        At each node split, a random subset of features ($\sqrt{M}$ for classification) is considered rather than all $M$ features, decorrelating the trees.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-check-to-slot text-cyan-400"></i> 3. Majority Voting & Feature Importance</span>
                        <span class="text-[10px] bg-cyan-500/20 text-cyan-300 px-2 py-0.5 rounded font-sans">Aggregation</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Predictions are aggregated via <strong>Majority Vote</strong>: $\hat{y} = \text{mode}(\{\hat{y}_1, \hat{y}_2, ..., \hat{y}_B\})$. Feature importance is calculated by averaging the Gini impurity decrease across all trees.
                    </p>
                </div>
            </div>

            <div class="bg-slate-900/60 p-4 rounded-lg border border-slate-700 text-xs text-slate-300 space-y-2">
                <h4 class="font-bold text-indigo-300 text-sm"><i class="fa-solid fa-circle-info text-amber-300"></i> Why Random Forest Outperforms Single Decision Trees</h4>
                <p class="text-slate-400 leading-relaxed">
                    Single decision trees have high variance and easily overfit noisy data. Random Forests reduce variance by averaging independent trees ($\text{Var}(\bar{X}) = \frac{\sigma^2}{B} + \frac{1-B}{B}\rho \sigma^2$). The combination of bootstrap data sampling and feature subsampling minimizes inter-tree correlation ($\rho$), resulting in lower generalization error.
                </p>
            </div>
        </div>

        function renderEnsembleUI(activeLevel = 0) {
            const data = datasets[scenario];
            const nTrees = parseInt(document.getElementById('nEstimatorsSelect').value) || 5;
            const treesContainer = document.getElementById('ensembleTreesContainer');
            const featContainer = document.getElementById('featureImportanceBox');

            document.getElementById('metricsBadge').innerText = `Trees: ${nTrees} | OOB Score: ${data.oobScore}%`;
            document.getElementById('valOobScore').innerText = activeLevel >= 3 ? `${data.oobScore}%` : "0.0%";
            document.getElementById('valTopFeature').innerText = activeLevel >= 3 ? data.features[0] : "--";

            if (activeLevel === 0) {
                treesContainer.innerHTML = `<div class="col-span-full p-4 text-center text-slate-500 italic bg-slate-950 rounded-lg border border-slate-700">Click 'Start Simulation' or 'Run / Execute All' to train ensemble trees.</div>`;
                featContainer.innerHTML = `<p class="text-slate-500 italic text-center">Start simulation to compute feature importances.</p>`;
                document.getElementById('valMajorityVote').innerText = "--";
                return;
            }

            // Render Individual Decision Trees Cards
            const selectedTrees = data.trees.slice(0, nTrees);
            let votesCount = { 0: 0, 1: 0 };

            let treesHtml = selectedTrees.map((t, idx) => {
                const isVoted = activeLevel >= 2;
                if (isVoted) votesCount[t.vote] += 1;

                return `
                    <div class="tree-card ${isVoted ? 'voted' : ''} bg-slate-900 border border-slate-700 p-2.5 rounded-lg text-center space-y-1">
                        <span class="text-amber-400 font-bold block text-[11px]"><i class="fa-solid fa-tree"></i> Tree #${idx + 1}</span>

                        </div>
                    </div>
                `;

            treesContainer.innerHTML = treesHtml;

                        <div>
                            <div class="flex justify-between text-[11px] mb-1">
                                <span class="text-slate-300 font-bold">${fName}</span>

                                <div class="feature-bar bg-indigo-500 h-full rounded-full" style="width: ${pct}%"></div>
                            </div>
                        </div>

                }).join('');
                featContainer.innerHTML = featHtml;
            } else {
                featContainer.innerHTML = `<p class="text-slate-500 italic text-center">Fitting ensemble decision trees...</p>`;
            }
        }

        function startSimulation() {
            updateDatasetDisplay();
            buildExecutionSteps();
            currentStepIdx = 0;

            hideResultBanner();
            clearConsole();

            const btnStep = document.getElementById('btnStep');
            btnStep.disabled = false;
            btnStep.className = "bg-emerald-600 hover:bg-emerald-500 text-white font-medium px-4 py-2 rounded-lg text-sm transition active:scale-95 shadow-lg shadow-emerald-600/20";

            appendConsole(`>>> Executing: ml_random_forest.py [Dataset: ${scenario}]`);
            executeStepAtIndex(0);
        }

        function executeStepAtIndex(idx) {
            const step = stepList[idx];
            highlightLine(step.line);

            renderEnsembleUI(step.activeLevel);

            appendConsole(step.consoleMsg);
            document.getElementById('explanationText').innerText = step.explanation;
            document.getElementById('currentStepBadge').innerText = `Epoch ${idx + 1}/${stepList.length}`;

            if (idx === stepList.length - 1) {
                const data = datasets[scenario];
                showResultBanner(`Sample Classified as ${data.classLabels[1]} (Majority Vote)`, false);
                const btnStep = document.getElementById('btnStep');
                btnStep.disabled = true;
                btnStep.className = "bg-slate-700 text-slate-400 font-medium px-4 py-2 rounded-lg text-sm transition cursor-not-allowed";
                document.getElementById('currentStepBadge').innerText = "Converged";
            }
        }

        function stepExecute() {
            if (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function runAll() {
            startSimulation();
            while (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function checkPretest() {
            const q1 = document.querySelector('input[name="pq1"]:checked')?.value;
            const q2 = document.querySelector('input[name="pq2"]:checked')?.value;
            let score = 0;

            const q1Box = document.getElementById('preq1-box');
            const q1fb = document.getElementById('preq1-feedback');
            q1fb.classList.remove('hidden');
            if (q1 === 'b') {
                score++;
                q1Box.className = "bg-slate-900 p-4 rounded-lg border border-emerald-500/50";
                q1fb.innerText = "✓ Correct! Random Forest combines Bagging (data sampling) and Random Subspace (feature sampling).";
                q1fb.className = "text-xs mt-2 font-medium text-emerald-400";
            } else {
                q1Box.className = "bg-slate-900 p-4 rounded-lg border border-rose-500/50";
                q1fb.innerText = "✗ Incorrect. Random Forests use Bagging and Random Feature Subsampling.";
                q1fb.className = "text-xs mt-2 font-medium text-rose-400";
            }

            const q2Box = document.getElementById('preq2-box');
            const q2fb = document.getElementById('preq2-feedback');
            q2fb.classList.remove('hidden');
            if (q2 === 'a') {
                score++;
                q2Box.className = "bg-slate-900 p-4 rounded-lg border border-emerald-500/50";
                q2fb.innerText = "✓ Correct! Classification Random Forests select max_features = sqrt(M) at each split.";
                q2fb.className = "text-xs mt-2 font-medium text-emerald-400";
            } else {
                q2Box.className = "bg-slate-900 p-4 rounded-lg border border-rose-500/50";
                q2fb.innerText = "✗ Incorrect. Default feature subset size for classification is sqrt(M).";
                q2fb.className = "text-xs mt-2 font-medium text-rose-400";
            }

            const badge = document.getElementById('pretest-score');
            badge.innerText = `Score: ${score} / 2`;
            badge.classList.remove('hidden');
        }

        function sendAiQuery() {
            const input = document.getElementById('aiQueryInput');
            const query = input.value.trim();
            if (!query) return;

            const chatBox = document.getElementById('aiChatBox');

            const userDiv = document.createElement('div');
            userDiv.className = 'flex items-start gap-3 justify-end';
            userDiv.innerHTML = `
                <div class="bg-indigo-600/30 border border-indigo-500/40 p-3 rounded-lg text-slate-200">
                    ${query}
                </div>
                <div class="w-7 h-7 rounded-full bg-slate-700 flex items-center justify-center text-slate-300 text-xs shrink-0">You</div>
            `;
            chatBox.appendChild(userDiv);
            input.value = '';

            let response = "Try testing different values of n_estimators or max_features in the simulator to see how individual tree predictions and feature importances update!";
            const qLower = query.toLowerCase();
            if (qLower.includes('bagging') || qLower.includes('bootstrap')) {
                response = "Bagging (Bootstrap Aggregating) trains each tree on a random sample drawn with replacement. The unselected ~36.8% samples form the Out-Of-Bag (OOB) set for validation!";
            } else if (qLower.includes('feature') || qLower.includes('importance') || qLower.includes('gini')) {
                response = "Feature Importance (`feature_importances_`) measures the total mean decrease in Gini impurity brought by splits on that feature across all trees in the forest!";
            } else if (qLower.includes('vote') || qLower.includes('majority') || qLower.includes('predict')) {
                response = "In classification, each tree outputs a class prediction vote. The Random Forest combines these votes and outputs the majority class as the final ensemble prediction!";
            }

            setTimeout(() => {
                const aiDiv = document.createElement('div');
                aiDiv.className = 'flex items-start gap-3';
                aiDiv.innerHTML = `
                    <div class="w-7 h-7 rounded-full bg-indigo-600 flex items-center justify-center text-white text-xs shrink-0">AI</div>
                    <div class="bg-slate-800 p-3 rounded-lg border border-slate-700 text-slate-200">
                        ${response}
                    </div>
                `;
                chatBox.appendChild(aiDiv);
                chatBox.scrollTop = chatBox.scrollHeight;
            }, 500);
        }
