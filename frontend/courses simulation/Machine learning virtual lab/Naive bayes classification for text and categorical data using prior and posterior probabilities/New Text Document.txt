        <!-- TAB 1: THEORY -->
        <div id="content-theory" class="tab-content bg-slate-800 border border-slate-700 rounded-xl p-6 shadow-md space-y-6">
            <h2 class="text-xl font-bold text-indigo-400 border-b border-slate-700 pb-2">Theory: Naïve Bayes Classification & Bayes' Theorem</h2>
            <p class="text-sm text-slate-300 leading-relaxed">
                <strong>Naïve Bayes</strong> is a probabilistic supervised learning algorithm based on <strong>Bayes' Theorem</strong>. It calculates the <strong>Posterior Probability</strong> of a class label $y \in \{C_1, C_2, ... C_k\}$ given a set of input feature vectors $X = (x_1, x_2, ... x_n)$. It is called <em>"Naïve"</em> because it makes the simplifying assumption that all input features are conditionally independent given the class label.
            </p>

            <div class="bg-slate-950 p-4 rounded-lg border border-slate-700 font-mono text-center text-indigo-300 text-sm space-y-1">
                <p>$P(Y | X) = \frac{P(X | Y) \cdot P(Y)}{P(X)} = \frac{P(Y) \cdot \prod_{i=1}^{n} P(x_i | Y)}{P(X)}$</p>
                <p class="text-xs text-amber-300">Posterior Probability $\propto$ Prior Probability $\times$ Likelihood Product</p>
            </div>

            <div class="grid grid-cols-1 md:grid-cols-3 gap-4 text-sm">
                
                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-chart-pie text-emerald-400"></i> 1. Prior Probability $P(Y)$</span>
                        <span class="text-[10px] bg-emerald-500/20 text-emerald-300 px-2 py-0.5 rounded font-sans">Class Odds</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        The baseline probability of observing class $Y$ in the training dataset prior to observing any input features $X$:
                        <br><code class="text-amber-300 font-mono">P(Y) = Count(Y) / Total Samples</code>.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-filter text-amber-400"></i> 2. Likelihood $P(X_i | Y)$</span>
                        <span class="text-[10px] bg-amber-500/20 text-amber-300 px-2 py-0.5 rounded font-sans">Feature Probability</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        The conditional probability of observing feature $x_i$ given class $Y$. For text classification, it represents the frequency of a word appearing in class documents.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-wand-magic-sparkles text-cyan-400"></i> 3. Laplace Smoothing ($\alpha$)</span>
                        <span class="text-[10px] bg-cyan-500/20 text-cyan-300 px-2 py-0.5 rounded font-sans">Zero-Count Fix</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Prevents zero-frequency problem ($P(x_i|Y)=0$ zeroing entire likelihood product) by adding pseudo-counts:
                        <br><code class="text-emerald-300 font-mono">P(x_i|Y) = (Count(x_i, Y) + α) / (Count(Y) + α|V|)</code>.
                    </p>
                </div>

            </div>

            <div class="bg-slate-900/60 p-4 rounded-lg border border-slate-700 text-xs text-slate-300 space-y-2">
                <h4 class="font-bold text-indigo-300 text-sm"><i class="fa-solid fa-circle-info text-amber-300"></i> Variants of Naïve Bayes Classifiers</h4>
                <ul class="list-disc list-inside space-y-1 text-slate-400 font-mono">
                    <li><strong class="text-slate-200 font-sans">MultinomialNB:</strong> Used for discrete word counts / term frequency text classification.</li>
                    <li><strong class="text-slate-200 font-sans">CategoricalNB:</strong> Used for categorical discrete features (e.g. Weather: Sunny, Rainy, Overcast).</li>
                    <li><strong class="text-slate-200 font-sans">GaussianNB:</strong> Used for continuous numerical features assuming Gaussian (normal) distribution.</li>
                </ul>
            </div>
        </div>

        function calculateBayesPosterior() {
            const queryRaw = document.getElementById('queryInput').value.toLowerCase();
            const tokens = queryRaw.split(/\s+/).filter(t => t.length > 0);
            const ds = datasets[scenario];
            const alpha = parseFloat(document.getElementById('alphaSelect').value) || 1.0;

            let logProd0 = Math.log(ds.priors[0]);
            let logProd1 = Math.log(ds.priors[1]);

            const tableRows = [];

            tokens.forEach(tok => {
                let p0 = ds.wordLikelihoods[tok] ? ds.wordLikelihoods[tok][0] : 0.05;
                let p1 = ds.wordLikelihoods[tok] ? ds.wordLikelihoods[tok][1] : 0.05;

                // Adjust for Laplace smoothing alpha
                if (alpha === 0 && !ds.wordLikelihoods[tok]) {
                    p0 = 1e-12; // Zero probability
                    p1 = 1e-12;
                }

                logProd0 += Math.log(p0);
                logProd1 += Math.log(p1);

                tableRows.push({ token: tok, p0, p1 });
            });

            // Exponentiate log sums safely
            const maxLog = Math.max(logProd0, logProd1);
            const unnorm0 = Math.exp(logProd0 - maxLog);
            const unnorm1 = Math.exp(logProd1 - maxLog);

            const post0 = unnorm0 / (unnorm0 + unnorm1);
            const post1 = unnorm1 / (unnorm0 + unnorm1);

            return { post0, post1, tableRows, predictedLabel: post1 > post0 ? 1 : 0 };
        }
