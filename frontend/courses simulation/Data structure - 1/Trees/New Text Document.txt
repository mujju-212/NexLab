    <script>
        function switchTab(t) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
            document.getElementById('content-' + t).classList.remove('hidden');
            const activeBtn = Array.from(document.querySelectorAll('.tab-btn')).find(b => b.innerText.toLowerCase().includes(t));
            if (activeBtn) activeBtn.classList.add('active');
        }
        async function runTraversal() {
            const nodes = ['nodeLeft', 'nodeRoot', 'nodeRight'];
            for(let n of nodes) {
                document.getElementById(n).classList.add('visited');
                await new Promise(r => setTimeout(r, 800));
            }
        }
        function insertBST() {
            const val = document.getElementById('bstInput').value;
            if(val) {
                document.getElementById('viz').innerHTML += `<div class="node bg-slate-700 w-12 h-12 flex items-center justify-center rounded-full font-bold">${val}</div>`;
                document.getElementById('bstInput').value = '';
            }
        }
        function askAI() {
            const q = document.getElementById('aiInput').value;
            document.getElementById('chatBox').innerHTML += `<p class="mb-2"><span class="text-indigo-400">You:</span> ${q}</p><p class="mb-2"><span class="text-emerald-400">AI:</span> Inorder traversal of a BST visits nodes in non-decreasing order.</p>`;
            document.getElementById('aiInput').value = '';
        }
    </script>
