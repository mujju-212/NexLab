    <script>
        let data = Array.from({length: 30}, () => Math.floor(Math.random() * 90) + 10);
        function switchTab(t) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
            document.getElementById('content-' + t).classList.remove('hidden');
            const activeBtn = Array.from(document.querySelectorAll('.tab-btn')).find(b => b.innerText.toLowerCase().includes(t));
            if (activeBtn) activeBtn.classList.add('active');
        }
        function renderViz() {
            const viz = document.getElementById('viz');
            viz.innerHTML = data.map(val => `<div class="bar bg-indigo-500 w-4" style="height:${val}%"></div>`).join('');
        }
        async function runSort() {
            const algo = document.getElementById('algoSelect').value;
            const speed = 105 - document.getElementById('speed').value;
            
            if (algo === 'bubble') {
                for(let i=0; i<data.length; i++) {
                    for(let j=0; j<data.length - i - 1; j++) {
                        if(data[j] > data[j+1]) {
                            [data[j], data[j+1]] = [data[j+1], data[j]];
                            renderViz();
                            await new Promise(r => setTimeout(r, speed));
                        }
                    }
                }
            } else if (algo === 'merge') {
                async function mergeSort(arr) {
                    if (arr.length <= 1) return arr;
                    let mid = Math.floor(arr.length / 2);
                    let left = await mergeSort(arr.slice(0, mid));
                    let right = await mergeSort(arr.slice(mid));
                    return merge(left, right);
                }
                async function merge(left, right) {
                    let sorted = [];
                    while(left.length && right.length) {
                        sorted.push(left[0] < right[0] ? left.shift() : right.shift());
                    }
                    data = [...sorted, ...left, ...right];
                    renderViz();
                    await new Promise(r => setTimeout(r, speed));
                    return [...sorted, ...left, ...right];
                }
                data = await mergeSort(data);
            } else if (algo === 'heap') {
                async function heapSort(arr) {
                    let n = arr.length;
                    for (let i = Math.floor(n/2)-1; i >= 0; i--) await heapify(arr, n, i);
                    for (let i = n-1; i > 0; i--) {
                        [arr[0], arr[i]] = [arr[i], arr[0]];
                        renderViz();
                        await new Promise(r => setTimeout(r, speed));
                        await heapify(arr, i, 0);
                    }
                }
                async function heapify(arr, n, i) {
                    let largest = i;
                    let l = 2*i + 1, r = 2*i + 2;
                    if (l < n && arr[l] > arr[largest]) largest = l;
                    if (r < n && arr[r] > arr[largest]) largest = r;
                    if (largest !== i) {
                        [arr[i], arr[largest]] = [arr[largest], arr[i]];
                        renderViz();
                        await new Promise(r => setTimeout(r, speed));
                        await heapify(arr, n, largest);
                    }
                }
                await heapSort(data);
            } else if (algo === 'quick') {
                async function quickSort(arr, left=0, right=arr.length-1) {
                    if(left < right) {
                        let pivot = await partition(arr, left, right);
                        await quickSort(arr, left, pivot-1);
                        await quickSort(arr, pivot+1, right);
                    }
                }
                async function partition(arr, left, right) {
                    let pivot = arr[right];
                    let i = left;
                    for(let j=left; j<right; j++) {
                        if(arr[j] < pivot) {
                            [arr[i], arr[j]] = [arr[j], arr[i]];
                            i++;
                            renderViz();
                            await new Promise(r => setTimeout(r, speed));
                        }
                    }
                    [arr[i], arr[right]] = [arr[right], arr[i]];
                    return i;
                }
                await quickSort(data);
            }
        }
        function resetData() { data = Array.from({length: 30}, () => Math.floor(Math.random() * 90) + 10); renderViz(); }
        function askAI() {
            const q = document.getElementById('aiInput').value;
            document.getElementById('chatBox').innerHTML += `<p class="mb-2"><span class="text-indigo-400">You:</span> ${q}</p><p class="mb-2"><span class="text-emerald-400">AI:</span> Bubble sort is $O(n^2)$, Merge and Quick sort are $O(n \log n)$, and Heap sort is $O(n \log n)$ while being in-place!</p>`;
            document.getElementById('aiInput').value = '';
        }
        renderViz();
