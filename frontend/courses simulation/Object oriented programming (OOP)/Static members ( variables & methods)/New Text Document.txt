    <script>
        const codes = {
            java: "class Counter { static int count = 0; }<br>Counter c1 = new Counter(); Counter.count++;",
            python: "class Counter: count = 0<br>c1 = Counter(); Counter.count += 1",
            cpp: "class Counter { public: static int count; };<br>int Counter::count = 0;<br>Counter c1; Counter::count++;"
        };
        let staticVal = 0;
        
        function switchTab(t) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active') || el.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
            document.getElementById('content-' + t).classList.add('active');
            document.getElementById('content-' + t).classList.remove('hidden');
            document.getElementById('btn-' + t).classList.add('active');
        }
        function loadSim(lang) { 
            document.getElementById('codeDisplay').innerHTML = codes[lang]; 
            document.querySelectorAll('.lang-btn').forEach(btn => btn.classList.remove('active'));
            document.getElementById('lang-' + lang).classList.add('active');
            resetSim();
        }
        function resetSim() { staticVal = 0; document.getElementById('heapDisplay').innerHTML = '<div class="p-4 border-2 border-slate-700 rounded-lg text-center text-slate-500 italic text-sm">Awaiting update...</div>'; }
        function modifyStatic() {
            staticVal++;
            document.getElementById('heapDisplay').innerHTML = `
                <div class="memory-slot bg-slate-800 p-4 rounded-lg border border-emerald-500/50">
                    <p class="text-[10px] text-emerald-400 font-bold mb-1">Execution Log</p>
                    <p class="text-sm font-bold text-white">Member: count</p>
                    <p class="text-xs text-slate-300">New Value: ${staticVal} (Shared across all instances)</p>
                </div>
            `;
        }
        function askAI() {
            const q = document.getElementById('aiInput').value;
            document.getElementById('chatBox').innerHTML += `<p class="mb-2"><span class="text-indigo-400">You:</span> ${q}</p><p class="mb-2"><span class="text-emerald-400">AI:</span> Static members are shared by all objects of the class.</p>`;
            document.getElementById('aiInput').value = '';
        }
        loadSim('java');
    </script>
