            <h2 class="text-2xl font-bold text-indigo-400 flex items-center gap-3"><i class="fa-solid fa-cube"></i> Theory: Classes & Objects</h2>
            <p class="text-sm text-slate-300 leading-relaxed">
                Object-Oriented Programming (OOP) is a paradigm based on "objects"—data structures containing data in the form of fields (attributes) and code in the form of procedures (methods).
            </p>
            <div class="grid grid-cols-1 md:grid-cols-2 gap-6">
                <div class="bg-slate-900 p-5 rounded-lg border border-slate-700">
                    <h4 class="font-bold text-indigo-300 mb-2">Class</h4>
                    <p class="text-xs text-slate-400">A class acts as a blueprint or template for creating objects. It defines the structure and behavior that its instances will possess.</p>
                </div>
                <div class="bg-slate-900 p-5 rounded-lg border border-slate-700">
                    <h4 class="font-bold text-indigo-300 mb-2">Object</h4>
                    <p class="text-xs text-slate-400">An object is an instance of a class. It represents a real-world entity and holds specific data values for the attributes defined in the class.</p>
                </div>
            </div>
            <h3 class="font-bold text-emerald-400 mt-4">Core Principles of OOP:</h3>
            <ul class="list-disc list-inside text-sm text-slate-300 space-y-2">
                <li><strong>Encapsulation:</strong> Bundling data and methods into a single unit (class).</li>
                <li><strong>Inheritance:</strong> Ability to create new classes based on existing ones.</li>
                <li><strong>Polymorphism:</strong> Ability for different classes to be treated as instances of a general class.</li>
                <li><strong>Abstraction:</strong> Hiding complex implementation details, showing only functionality.</li>
            </ul>
        </div>

        function switchTab(t) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active') || el.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
            document.getElementById('content-' + t).classList.add('active');
            document.getElementById('content-' + t).classList.remove('hidden');
            document.getElementById('btn-' + t).classList.add('active');
        }
        function loadSim(lang) { document.getElementById('codeDisplay').innerHTML = codes[lang]; }
        function instantiateObject() {
            const heap = document.getElementById('heapDisplay');
            heap.innerHTML = `
                <div class="memory-slot bg-slate-800 p-4 rounded-lg border border-indigo-500/50">
                    <p class="text-[10px] text-indigo-400 font-bold mb-1">Heap Allocation @ 0x4A2F</p>
                    <p class="text-sm font-bold text-emerald-400">Class: Student</p>
                    <p class="text-xs text-slate-300">name: "Alice"</p>
                </div>
            `;
        }
        function askAI() {
            const q = document.getElementById('aiInput').value;
            document.getElementById('chatBox').innerHTML += `<p class="mb-2"><span class="text-indigo-400">You:</span> ${q}</p><p class="mb-2"><span class="text-emerald-400">AI:</span> A class is a template, while an object is a specific instance allocated in Heap memory.</p>`;
            document.getElementById('aiInput').value = '';
        }
