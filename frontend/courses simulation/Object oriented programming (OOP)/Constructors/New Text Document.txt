    <script>
        const codes = {
            java: "public class Student {<br>&nbsp;&nbsp;String name;<br>&nbsp;&nbsp;public Student(String n) { name = n; }<br>}<br>Student s1 = new Student(\"Alice\");",
            python: "class Student:<br>&nbsp;&nbsp;def __init__(self, name): self.name = name<br>s1 = Student('Alice')",
            cpp: "class Student {<br>&nbsp;&nbsp;public: string name;<br>&nbsp;&nbsp;Student(string n) : name(n) {}<br>};<br>Student s1(\"Alice\");"
        };
        let currentLang = 'java';

        function switchTab(t) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active') || el.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
            document.getElementById('content-' + t).classList.add('active');
            document.getElementById('content-' + t).classList.remove('hidden');
            document.getElementById('btn-' + t).classList.add('active');
        }
        function loadSim(lang) { 
            currentLang = lang;
            document.getElementById('codeDisplay').innerHTML = codes[lang]; 
            document.querySelectorAll('button[onclick^="loadSim"]').forEach(btn => btn.classList.remove('font-bold', 'text-indigo-400'));
            document.getElementById('lang-' + lang).classList.add('font-bold', 'text-indigo-400');
        }
        function instantiateObject() {
            const display = document.getElementById('heapDisplay');
            const addr = Math.floor(Math.random() * 9000) + 1000;
            display.innerHTML = `
                <div class="memory-slot bg-slate-800 p-4 rounded-lg border border-emerald-500/50 shadow-inner">
                    <p class="text-[10px] text-emerald-400 font-bold mb-1">Heap Allocation @ 0x${addr.toString(16).toUpperCase()}</p>
                    <p class="text-sm font-bold text-white">Instance: s1 (${currentLang.toUpperCase()})</p>
                    <p class="text-xs text-slate-300">Initialized by Constructor</p>
                </div>
            `;
        }
        function resetHeap() {
            document.getElementById('heapDisplay').innerHTML = `<div class="p-4 border-2 border-slate-700 rounded-lg text-center text-slate-500 italic text-sm">No instances allocated yet.</div>`;
        }
        function askAI() {
            const q = document.getElementById('aiInput').value;
            document.getElementById('chatBox').innerHTML += `<p class="mb-2"><span class="text-indigo-400">You:</span> ${q}</p><p class="mb-2"><span class="text-emerald-400">AI:</span> A constructor initializes object attributes at the moment of object creation in heap memory.</p>`;
            document.getElementById('aiInput').value = '';
        }
        loadSim('java');
    </script>
