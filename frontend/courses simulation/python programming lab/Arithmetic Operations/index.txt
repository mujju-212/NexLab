    <script>
        // Execution State Variables
        let currentStep = 0;
        let valA = 10;
        let valB = 3;
        let op = '+';
        let resultC = 0;
        let typeCStr = 'int';

        // Keyboard Shortcut Listener for Simulation Inputs
        document.addEventListener("DOMContentLoaded", () => {
            const inputs = ['inputA', 'inputB'];
            inputs.forEach(id => {
                document.getElementById(id).addEventListener('keydown', (e) => {
                    if (e.key === 'Enter') runAll();
                });
            });
            updateOperatorDisplay();
        });

        // UI Helper Functions
        function highlightLine(lineNum) {
            document.querySelectorAll('.code-line').forEach(el => el.classList.remove('active'));
            if (lineNum > 0) {
                const target = document.getElementById(`line${lineNum}`);
                if (target) target.classList.add('active');
            }
        }

        function clearMemory() {
            document.getElementById('valA').innerText = '--';
            document.getElementById('valB').innerText = '--';
            document.getElementById('valC').innerText = '--';
            document.getElementById('typeC').innerText = '--';
            ['memA', 'memB', 'memC'].forEach(id => {
                const el = document.getElementById(id);
                if (el) el.classList.remove('updated');
            });
        }

        function clearConsole() {
            document.getElementById('terminal').innerHTML = '';
        }

        function appendConsole(htmlContent) {
            const term = document.getElementById('terminal');
            const p = document.createElement('p');
            p.innerHTML = htmlContent;
            term.appendChild(p);
            term.scrollTop = term.scrollHeight;
        }

        function triggerMemUpdate(memId) {
            const el = document.getElementById(memId);
            if (!el) return;
            el.classList.add('updated');
            setTimeout(() => { el.classList.remove('updated'); }, 500);
        }

        function switchTab(tabName) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(el => {
                el.classList.remove('text-indigo-400', 'bg-slate-700/60', 'border', 'border-slate-600');
                el.classList.add('text-slate-400');
            });

            document.getElementById(`content-${tabName}`).classList.remove('hidden');
            const activeBtn = document.getElementById(`tab-${tabName}`);
            activeBtn.classList.remove('text-slate-400');
            activeBtn.classList.add('text-indigo-400', 'bg-slate-700/60', 'border', 'border-slate-600');
        }

        // Quick Preset Loader Button Handler
        function loadPreset(a, operator, b) {
            document.getElementById('inputA').value = a;
            document.getElementById('inputOp').value = operator;
            document.getElementById('inputB').value = b;
            updateOperatorDisplay();
            runAll();
        }

        function updateOperatorDisplay() {
            valA = parseFloat(document.getElementById('inputA').value) || 0;
            valB = parseFloat(document.getElementById('inputB').value) || 0;
            op = document.getElementById('inputOp').value;

            document.getElementById('codeValA').innerText = valA.toFixed(1);
            document.getElementById('codeValB').innerText = valB.toFixed(1);
            document.getElementById('codeOp').innerText = op;
        }

        function startSimulation() {
            valA = parseFloat(document.getElementById('inputA').value);
            valB = parseFloat(document.getElementById('inputB').value);
            if (isNaN(valA)) valA = 0;
            if (isNaN(valB)) valB = 0;
            op = document.getElementById('inputOp').value;

            updateOperatorDisplay();

            if ((op === '/' || op === '%' || op === '//') && valB === 0) {
                clearConsole();
                appendConsole("<span class='text-rose-400 font-bold'>ZeroDivisionError: division or modulus by zero!</span>");
                document.getElementById('explanationText').innerText = "Error: Division or modulus by zero raises ZeroDivisionError in Python.";
                showResultBanner("Error: ZeroDivisionError", true);
                currentStep = 0;
                document.getElementById('btnStep').disabled = true;
                document.getElementById('btnStep').className = "bg-slate-700 text-slate-400 font-medium px-4 py-2 rounded-lg text-sm transition cursor-not-allowed";
                return;
            }

            // Calculate Result accurately
            switch(op) {
                case '+': resultC = valA + valB; typeCStr = (Number.isInteger(resultC) ? 'int' : 'float'); break;
                case '-': resultC = valA - valB; typeCStr = (Number.isInteger(resultC) ? 'int' : 'float'); break;
                case '*': resultC = valA * valB; typeCStr = (Number.isInteger(resultC) ? 'int' : 'float'); break;
                case '/': resultC = valA / valB; typeCStr = 'float'; break;
                case '%': resultC = valA % valB; typeCStr = (Number.isInteger(resultC) ? 'int' : 'float'); break;
                case '**': resultC = Math.pow(valA, valB); typeCStr = (Number.isInteger(resultC) ? 'int' : 'float'); break;
                case '//': resultC = Math.floor(valA / valB); typeCStr = (Number.isInteger(valA) && Number.isInteger(valB)) ? 'int' : 'float'; break;
            }

            if (typeCStr === 'float' && !Number.isInteger(resultC)) {
                resultC = parseFloat(resultC.toFixed(4));
            }

            currentStep = 1;
            const btnStep = document.getElementById('btnStep');
            btnStep.disabled = false;
            btnStep.className = "bg-emerald-600 hover:bg-emerald-500 text-white font-medium px-4 py-2 rounded-lg text-sm transition active:scale-95 shadow-lg shadow-emerald-600/20";
            
            hideResultBanner();
            highlightLine(1);
            clearMemory();
            clearConsole();
            appendConsole(`>>> Executing: script.py [a = ${valA}, b = ${valB}, op = '${op}']`);
            document.getElementById('explanationText').innerText = "Line 1 executing: Storing variable 'a'. Click 'Step Execution' or 'Run / Execute All'.";
            document.getElementById('currentStepBadge').innerText = "Step 1/4";
        }

        function stepExecute() {
            if (currentStep === 1) {
                document.getElementById('valA').innerText = valA;
                triggerMemUpdate('memA');
                highlightLine(2);
                appendConsole(`> a = ${valA}`);
                document.getElementById('explanationText').innerText = `Variable 'a' stored (${valA}). Executing Line 2 for variable 'b'.`;
                currentStep = 2;
                document.getElementById('currentStepBadge').innerText = "Step 2/4";
            } else if (currentStep === 2) {
                document.getElementById('valB').innerText = valB;
                triggerMemUpdate('memB');
                highlightLine(3);
                appendConsole(`> b = ${valB}`);
                document.getElementById('explanationText').innerText = `Variable 'b' stored (${valB}). Line 3 evaluates expression: c = ${valA} ${op} ${valB}.`;
                currentStep = 3;
                document.getElementById('currentStepBadge').innerText = "Step 3/4";
            } else if (currentStep === 3) {
                document.getElementById('valC').innerText = resultC;
                document.getElementById('typeC').innerText = typeCStr;
                triggerMemUpdate('memC');
                highlightLine(4);
                appendConsole(`> c = ${valA} ${op} ${valB}  -->  c = ${resultC}`);
                document.getElementById('explanationText').innerText = `Expression evaluated! c = ${resultC} (type: ${typeCStr}). Executing print statement.`;
                currentStep = 4;
                document.getElementById('currentStepBadge').innerText = "Step 4/4";
            } else if (currentStep === 4) {
                appendConsole(`<span class='text-emerald-300 font-bold'>Result c = ${resultC}</span>`);
                showResultBanner(`Result c = ${resultC} (${typeCStr})`, false);
                highlightLine(0);
                document.getElementById('explanationText').innerText = "Execution completed! Final output displayed below.";
                
                const btnStep = document.getElementById('btnStep');
                btnStep.disabled = true;
                btnStep.className = "bg-slate-700 text-slate-400 font-medium px-4 py-2 rounded-lg text-sm transition cursor-not-allowed";
                document.getElementById('currentStepBadge').innerText = "Finished";
                currentStep = 5; // Finish state
            }
        }

        function runAll() {
            startSimulation();
            if ((op === '/' || op === '%' || op === '//') && valB === 0) return;
            while (currentStep >= 1 && currentStep <= 4) {
                stepExecute();
            }
        }

        function showResultBanner(text, isError) {
            const banner = document.getElementById('resultBanner');
            const bannerText = document.getElementById('resultBannerText');
            const bannerIcon = document.getElementById('bannerIcon');
            const bannerStatus = document.getElementById('bannerStatus');
            
            bannerText.innerText = text;
            if (isError) {
                bannerText.className = "text-lg font-mono font-bold text-rose-400";
                bannerIcon.className = "w-10 h-10 rounded-full bg-rose-500/20 text-rose-400 flex items-center justify-center text-lg font-bold";
                bannerIcon.innerHTML = `<i class="fa-solid fa-triangle-exclamation"></i>`;
                bannerStatus.innerText = "Status: Error";
                bannerStatus.className = "text-xs font-mono text-rose-400 bg-rose-950/80 px-3 py-1 rounded-md border border-rose-800";
            } else {
                bannerText.className = "text-xl font-mono font-bold text-emerald-300";
                bannerIcon.className = "w-10 h-10 rounded-full bg-emerald-500/20 text-emerald-400 flex items-center justify-center text-lg font-bold";
                bannerIcon.innerHTML = `<i class="fa-solid fa-square-check"></i>`;
                bannerStatus.innerText = "Status: OK";
                bannerStatus.className = "text-xs font-mono text-emerald-400 bg-slate-900/80 px-3 py-1 rounded-md border border-slate-700";
            }
            banner.classList.remove('hidden');
        }

        function hideResultBanner() {
            document.getElementById('resultBanner').classList.add('hidden');
        }

        function sendAiQuery() {
            const input = document.getElementById('aiQueryInput');
            const query = input.value.trim();
            if (!query) return;

            const chatBox = document.getElementById('aiChatBox');

            const userDiv = document.createElement('div');
            userDiv.className = 'flex items-start gap-3 justify-end';
            userDiv.innerHTML = `
                <div class="bg-indigo-600/30 border border-indigo-500/40 p-3 rounded-lg text-slate-200">
                    ${query}
                </div>
                <div class="w-7 h-7 rounded-full bg-slate-700 flex items-center justify-center text-slate-300 text-xs shrink-0">You</div>
            `;
            chatBox.appendChild(userDiv);
            input.value = '';

            let response = "Try running that exact operation in the simulator and check the output data type in variable memory 'c'!";
            const qLower = query.toLowerCase();
            if (qLower.includes('//') || qLower.includes('floor')) {
                response = "Floor division (`//`) discards the fractional decimal part. If you divide 7 by 2 normally you get 3.5. What integer lies immediately below 3.5?";
            } else if (qLower.includes('/') || qLower.includes('float')) {
                response = "In Python 3, standard division (`/`) always returns a floating-point number, even if the result is a whole integer. Why do you think that consistency is helpful?";
            } else if (qLower.includes('%') || qLower.includes('modulus') || qLower.includes('remainder')) {
                response = "The modulus operator (`%`) calculates the remainder after division. For example, 10 divided by 3 is 3 with a remainder of 1. How could this be useful for checking even/odd numbers?";
            } else if (qLower.includes('**') || qLower.includes('power') || qLower.includes('exponent')) {
                response = "Exponentiation (`**`) raises the left operand to the power of the right operand (e.g., 2 ** 3 = 8). Try testing `2 ** 8` using the preset buttons!";
            }

            setTimeout(() => {
                const aiDiv = document.createElement('div');
                aiDiv.className = 'flex items-start gap-3';
                aiDiv.innerHTML = `
                    <div class="w-7 h-7 rounded-full bg-indigo-600 flex items-center justify-center text-white text-xs shrink-0">AI</div>
                    <div class="bg-slate-800 p-3 rounded-lg border border-slate-700 text-slate-200">
                        ${response}
                    </div>
                `;
                chatBox.appendChild(aiDiv);
                chatBox.scrollTop = chatBox.scrollHeight;
            }, 500);
        }
