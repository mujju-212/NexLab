            <h2 class="text-xl font-bold text-indigo-400 border-b border-slate-700 pb-2">Theory: Iteration Structures in Python</h2>
            <p class="text-sm text-slate-300 leading-relaxed">
                In programming, <strong>looping</strong> or iteration allows a block of statements to be executed repeatedly until a specified condition is met. Python provides two primary loop constructs: <code class="bg-slate-900 text-indigo-300 px-1.5 py-0.5 rounded font-mono">for</code> loops (definite iteration) and <code class="bg-slate-900 text-indigo-300 px-1.5 py-0.5 rounded font-mono">while</code> loops (indefinite iteration).
            </p>

            <div class="grid grid-cols-1 md:grid-cols-2 gap-4 text-sm">
                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono"><i class="fa-solid fa-rotate-right text-emerald-400"></i> For Loop & range()</h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        The <code class="text-indigo-300">for</code> loop iterates over a sequence generated by <code class="text-amber-300">range(start, stop, step)</code>.
                    </p>
                    <div class="bg-slate-950 p-2.5 rounded font-mono text-xs text-slate-300 border border-slate-800">
                        total = 0<br>
                        for i in range(1, 6):<br>
                        &nbsp;&nbsp;&nbsp;&nbsp;total += i
                    </div>
                    <p class="text-[11px] text-slate-400 italic">Executes exactly 5 times ($i = 1, 2, 3, 4, 5$). Note that <code class="text-slate-200">stop</code> (6) is exclusive!</p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono"><i class="fa-solid fa-arrows-spin text-amber-400"></i> While Loop</h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        The <code class="text-indigo-300">while</code> loop continues executing as long as its Boolean condition remains <code class="text-emerald-400">True</code>.
                    </p>
                    <div class="bg-slate-950 p-2.5 rounded font-mono text-xs text-slate-300 border border-slate-800">
                        count = 1<br>
                        while count &lt;= 4:<br>
                        &nbsp;&nbsp;&nbsp;&nbsp;count += 1
                    </div>
                    <p class="text-[11px] text-slate-400 italic">Care must be taken to update the loop counter variable to avoid infinite loops.</p>
                </div>
            </div>

            <div class="bg-slate-900/60 p-4 rounded-lg border border-slate-700 text-xs text-slate-300 space-y-2">
                <h4 class="font-bold text-indigo-300 text-sm"><i class="fa-solid fa-lightbulb text-amber-300"></i> Key Architectural Concepts</h4>
                <ul class="list-disc list-inside space-y-1 text-slate-400">
                    <li><strong class="text-slate-200">Loop Variable:</strong> Keeps track of current iteration state ($i$, $count$).</li>
                    <li><strong class="text-slate-200">Accumulator:</strong> Variable used to collect or sum running values across iterations ($total$).</li>
                    <li><strong class="text-slate-200">Loop Termination:</strong> Occurs when sequence completes or boolean evaluation returns <code class="text-rose-400">False</code>.</li>
                </ul>
            </div>
        </div>

    <script>
        // Execution State Variables
        let scenario = 'sum';
        let limitN = 5;
        let stepList = [];
        let currentStepIdx = 0;

        document.addEventListener("DOMContentLoaded", () => {

                if (e.key === 'Enter') runAll();

        function switchTab(tabName) {

        function loadPreset(presetScenario, nVal) {

        function updateScenarioDisplay() {
            scenario = document.getElementById('loopScenario').value;
            limitN = parseInt(document.getElementById('inputN').value) || 5;
            if (limitN < 1) limitN = 1;
            if (limitN > 10) limitN = 10;

            const codeContainer = document.getElementById('codeContainer');
            if (scenario === 'sum') {
                codeContainer.innerHTML = `
                    <div id="line1" class="code-line px-3 py-1.5 rounded flex items-center gap-4 text-slate-300"><span class="text-slate-600 text-xs w-4">1</span><span>total = 0</span></div>
                    <div id="line2" class="code-line px-3 py-1.5 rounded flex items-center gap-4 text-slate-300"><span class="text-slate-600 text-xs w-4">2</span><span>for i in range(1, ${limitN + 1}):</span></div>
                    <div id="line3" class="code-line px-3 py-1.5 rounded flex items-center gap-4 text-slate-300"><span class="text-slate-600 text-xs w-4">3</span><span>&nbsp;&nbsp;&nbsp;&nbsp;total = total + i</span></div>
                    <div id="line4" class="code-line px-3 py-1.5 rounded flex items-center gap-4 text-slate-300"><span class="text-slate-600 text-xs w-4">4</span><span>print(<span class="text-emerald-300">f"Final Total = {total}"</span>)</span></div>
                `;
            } else if (scenario === 'factorial') {
                codeContainer.innerHTML = `
                    <div id="line1" class="code-line px-3 py-1.5 rounded flex items-center gap-4 text-slate-300"><span class="text-slate-600 text-xs w-4">1</span><span>fact = 1</span></div>
                    <div id="line2" class="code-line px-3 py-1.5 rounded flex items-center gap-4 text-slate-300"><span class="text-slate-600 text-xs w-4">2</span><span>for i in range(1, ${limitN + 1}):</span></div>
                    <div id="line3" class="code-line px-3 py-1.5 rounded flex items-center gap-4 text-slate-300"><span class="text-slate-600 text-xs w-4">3</span><span>&nbsp;&nbsp;&nbsp;&nbsp;fact = fact * i</span></div>
                    <div id="line4" class="code-line px-3 py-1.5 rounded flex items-center gap-4 text-slate-300"><span class="text-slate-600 text-xs w-4">4</span><span>print(<span class="text-emerald-300">f"{${limitN}}! = {fact}"</span>)</span></div>
                `;
            } else if (scenario === 'while_count') {
                codeContainer.innerHTML = `
                    <div id="line1" class="code-line px-3 py-1.5 rounded flex items-center gap-4 text-slate-300"><span class="text-slate-600 text-xs w-4">1</span><span>count = 1</span></div>
                    <div id="line2" class="code-line px-3 py-1.5 rounded flex items-center gap-4 text-slate-300"><span class="text-slate-600 text-xs w-4">2</span><span>while count &lt;= ${limitN}:</span></div>
                    <div id="line3" class="code-line px-3 py-1.5 rounded flex items-center gap-4 text-slate-300"><span class="text-slate-600 text-xs w-4">3</span><span>&nbsp;&nbsp;&nbsp;&nbsp;count += 1</span></div>
                    <div id="line4" class="code-line px-3 py-1.5 rounded flex items-center gap-4 text-slate-300"><span class="text-slate-600 text-xs w-4">4</span><span>print(<span class="text-emerald-300">f"Reached limit {count - 1}"</span>)</span></div>
                `;
            } else if (scenario === 'even_sum') {
                codeContainer.innerHTML = `
                    <div id="line1" class="code-line px-3 py-1.5 rounded flex items-center gap-4 text-slate-300"><span class="text-slate-600 text-xs w-4">1</span><span>even_sum = 0</span></div>
                    <div id="line2" class="code-line px-3 py-1.5 rounded flex items-center gap-4 text-slate-300"><span class="text-slate-600 text-xs w-4">2</span><span>for i in range(2, ${limitN + 1}, 2):</span></div>
                    <div id="line3" class="code-line px-3 py-1.5 rounded flex items-center gap-4 text-slate-300"><span class="text-slate-600 text-xs w-4">3</span><span>&nbsp;&nbsp;&nbsp;&nbsp;even_sum += i</span></div>
                    <div id="line4" class="code-line px-3 py-1.5 rounded flex items-center gap-4 text-slate-300"><span class="text-slate-600 text-xs w-4">4</span><span>print(<span class="text-emerald-300">f"Sum Evens = {even_sum}"</span>)</span></div>
                `;
            }
        }

        // Generate Step-by-Step Execution Sequence
        function buildExecutionSteps() {
            stepList = [];
            
            // Step 0: Line 1 Initialization
            let accVal = (scenario === 'factorial') ? 1 : ((scenario === 'while_count') ? 1 : 0);
            stepList.push({
                line: 1,
                iVal: '--',
                nVal: limitN,
                accVal: accVal,
                consoleMsg: `> Initialized variables [Limit N = ${limitN}]`,
                explanation: `Line 1 executing: Storing initial accumulator state (${accVal}).`
            });

            if (scenario === 'sum') {
                for (let i = 1; i <= limitN; i++) {
                    stepList.push({
                        line: 2,
                        iVal: i,
                        nVal: limitN,
                        accVal: accVal,
                        consoleMsg: `> Loop Iteration i = ${i} (Condition ${i} <= ${limitN} True)`,
                        explanation: `Line 2: Loop condition holds True for i = ${i}.`
                    });
                    accVal += i;
                    stepList.push({
                        line: 3,
                        iVal: i,
                        nVal: limitN,
                        accVal: accVal,
                        consoleMsg: `  └─ total = total + ${i} --> total = ${accVal}`,
                        explanation: `Line 3: Adding ${i} to total. Accumulator updated to ${accVal}.`
                    });
                }
            } else if (scenario === 'factorial') {
                for (let i = 1; i <= limitN; i++) {
                    stepList.push({
                        line: 2,
                        iVal: i,
                        nVal: limitN,
                        accVal: accVal,
                        consoleMsg: `> Factorial Iteration i = ${i}`,
                        explanation: `Line 2: Evaluating loop header for i = ${i}.`
                    });
                    accVal *= i;
                    stepList.push({
                        line: 3,
                        iVal: i,
                        nVal: limitN,
                        accVal: accVal,
                        consoleMsg: `  └─ fact = fact * ${i} --> fact = ${accVal}`,
                        explanation: `Line 3: Multiplying fact by ${i}. Accumulator updated to ${accVal}.`
                    });
                }
            } else if (scenario === 'while_count') {
                let count = 1;
                while (count <= limitN) {
                    stepList.push({
                        line: 2,
                        iVal: count,
                        nVal: limitN,
                        accVal: count,
                        consoleMsg: `> While Condition (count ${count} <= ${limitN}) True`,
                        explanation: `Line 2: Checking while condition (${count} <= ${limitN}).`
                    });
                    count++;
                    stepList.push({
                        line: 3,
                        iVal: count - 1,
                        nVal: limitN,
                        accVal: count,
                        consoleMsg: `  └─ count += 1 --> count = ${count}`,
                        explanation: `Line 3: Incremented counter variable to ${count}.`
                    });
                }
                accVal = count - 1;
            } else if (scenario === 'even_sum') {
                for (let i = 2; i <= limitN; i += 2) {
                    stepList.push({
                        line: 2,
                        iVal: i,
                        nVal: limitN,
                        accVal: accVal,
                        consoleMsg: `> Even Step Iteration i = ${i}`,
                        explanation: `Line 2: Loop header step = 2, i = ${i}.`
                    });
                    accVal += i;
                    stepList.push({
                        line: 3,
                        iVal: i,
                        nVal: limitN,
                        accVal: accVal,
                        consoleMsg: `  └─ even_sum += ${i} --> even_sum = ${accVal}`,
                        explanation: `Line 3: Added even integer ${i}. Accumulator = ${accVal}.`
                    });
                }
            }

            // Final Step: Line 4 Print & Completion
            stepList.push({
                line: 4,
                iVal: 'Done',
                nVal: limitN,
                accVal: accVal,
                consoleMsg: `<span class='text-emerald-300 font-bold'>Final Loop Output = ${accVal}</span>`,
                explanation: `Loop execution finished! Final accumulated output: ${accVal}.`
            });
        }

        function startSimulation() {
            updateScenarioDisplay();
            buildExecutionSteps();
            currentStepIdx = 0;

            hideResultBanner();
            clearMemory();
            clearConsole();
            
            const btnStep = document.getElementById('btnStep');
            btnStep.disabled = false;
            btnStep.className = "bg-emerald-600 hover:bg-emerald-500 text-white font-medium px-4 py-2 rounded-lg text-sm transition active:scale-95 shadow-lg shadow-emerald-600/20";

            appendConsole(`>>> Executing: loop_script.py [Scenario: ${scenario}, Limit: ${limitN}]`);
            executeStepAtIndex(0);
        }

        function executeStepAtIndex(idx) {
            const step = stepList[idx];
            highlightLine(step.line);

            document.getElementById('valI').innerText = step.iVal;
            document.getElementById('valN').innerText = step.nVal;
            document.getElementById('valAcc').innerText = step.accVal;

            if (step.line === 3) {
                triggerMemUpdate('memAcc');
            } else if (step.line === 2) {
                triggerMemUpdate('memI');
            }

            appendConsole(step.consoleMsg);
            document.getElementById('explanationText').innerText = step.explanation;
            document.getElementById('currentStepBadge').innerText = `Step ${idx + 1}/${stepList.length}`;

            if (step.line === 4) {
                showResultBanner(`Final Output = ${step.accVal}`, false);
                const btnStep = document.getElementById('btnStep');
                btnStep.disabled = true;
                btnStep.className = "bg-slate-700 text-slate-400 font-medium px-4 py-2 rounded-lg text-sm transition cursor-not-allowed";
                document.getElementById('currentStepBadge').innerText = "Finished";
            }
        }

        function stepExecute() {
            if (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function runAll() {
            startSimulation();
            while (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function resetSimulation() {
            currentStepIdx = 0;
            highlightLine(0);
            clearMemory();
            clearConsole();
            hideResultBanner();
            appendConsole("<p class='text-slate-500 italic'>>>> Click 'Run / Execute All' or 'Start Simulation' to execute.</p>");
            document.getElementById('explanationText').innerText = "Select parameters and click 'Start Simulation' or 'Run / Execute All' to execute.";
            document.getElementById('currentStepBadge').innerText = "Step: Ready";

            const btnStep = document.getElementById('btnStep');
            btnStep.disabled = true;
            btnStep.className = "bg-slate-700 text-slate-400 font-medium px-4 py-2 rounded-lg text-sm transition cursor-not-allowed";
        }

        function highlightLine(lineNum) {
            document.querySelectorAll('.code-line').forEach(el => el.classList.remove('active'));
            if (lineNum > 0) {
                const target = document.getElementById(`line${lineNum}`);
                if (target) target.classList.add('active');
            }
        }

        function clearMemory() {

        }

        }

        function sendAiQuery() {
            const input = document.getElementById('aiQueryInput');
            const query = input.value.trim();
            if (!query) return;

            const chatBox = document.getElementById('aiChatBox');

            const userDiv = document.createElement('div');
            userDiv.className = 'flex items-start gap-3 justify-end';
            userDiv.innerHTML = `
                <div class="bg-indigo-600/30 border border-indigo-500/40 p-3 rounded-lg text-slate-200">
                    ${query}
                </div>
                <div class="w-7 h-7 rounded-full bg-slate-700 flex items-center justify-center text-slate-300 text-xs shrink-0">You</div>
            `;
            chatBox.appendChild(userDiv);
            input.value = '';

            let response = "Try testing that loop scenario in the simulator and watch how the RAM memory counter updates on every pass!";
            const qLower = query.toLowerCase();
            if (qLower.includes('range') || qLower.includes('exclusive')) {
                response = "The `range(start, stop)` function generates numbers up to but NOT including `stop`. If you want to include 5, what should your stop parameter be?";
            } else if (qLower.includes('while') || qLower.includes('infinite')) {
                response = "A `while` loop checks a boolean condition before every pass. What happens to the loop condition if you forget to increment your counter inside the loop body?";
            } else if (qLower.includes('for') || qLower.includes('loop')) {
                response = "`for` loops in Python iterate over sequences automatically. Notice in our memory inspector how `i` changes state automatically with every pass!";
            }

            setTimeout(() => {
                const aiDiv = document.createElement('div');
                aiDiv.className = 'flex items-start gap-3';
                aiDiv.innerHTML = `
                    <div class="w-7 h-7 rounded-full bg-indigo-600 flex items-center justify-center text-white text-xs shrink-0">AI</div>
                    <div class="bg-slate-800 p-3 rounded-lg border border-slate-700 text-slate-200">
                        ${response}
                    </div>
                `;
                chatBox.appendChild(aiDiv);
                chatBox.scrollTop = chatBox.scrollHeight;
            }, 500);
        }
