        function renderHeapObjects(objectsList) {
            const container = document.getElementById('heapObjectsContainer');
            if (!objectsList || objectsList.length === 0) {
                container.innerHTML = `
                    <div class="col-span-2 text-slate-500 italic text-xs p-4 text-center border border-dashed border-slate-700 rounded-lg">
                        Select scenario and click 'Start Simulation' to view constructor allocation & inheritance in Heap RAM.
                    </div>
                `;
                return;
            }

            let html = '';
            objectsList.forEach(obj => {
                html += `
                    <div class="object-box active-obj bg-slate-900 border-2 border-indigo-500/60 p-3 rounded-lg font-mono space-y-1.5 shadow-md">
                        <div class="flex justify-between items-center border-b border-slate-700 pb-1 text-xs">
                            <span class="font-bold text-amber-400">${obj.varName} <span class="text-slate-400 font-normal">(${obj.className})</span></span>
                            <span class="text-[9px] bg-slate-800 text-emerald-400 px-1.5 py-0.5 rounded border border-slate-700">${obj.addr}</span>
                        </div>
                        <div class="text-xs space-y-1 pt-1">
                `;

                for (const [key, val] of Object.entries(obj.attrs)) {
                    html += `
                        <div class="flex justify-between items-center bg-slate-950 px-2 py-1 rounded text-[11px] border border-slate-800">
                            <span class="text-indigo-300">self.${key}:</span>
                            <span class="text-emerald-300 font-bold">${val}</span>
                        </div>
                    `;
                }

                html += `
                        </div>
                    </div>
                `;
            });

            container.innerHTML = html;
        }

        function startSimulation() {
            updateScenarioDisplay();
            buildExecutionSteps();
            currentStepIdx = 0;

            hideResultBanner();
            clearConsole();

            const btnStep = document.getElementById('btnStep');
            btnStep.disabled = false;
            btnStep.className = "bg-emerald-600 hover:bg-emerald-500 text-white font-medium px-4 py-2 rounded-lg text-sm transition active:scale-95 shadow-lg shadow-emerald-600/20";

            appendConsole(`>>> Executing: constructor_inheritance_demo.py [Scenario: ${scenario}]`);
            executeStepAtIndex(0);
        }

        function executeStepAtIndex(idx) {
            const step = stepList[idx];
            highlightLine(step.line);

            renderHeapObjects(step.objects);

            appendConsole(step.consoleMsg);
            document.getElementById('explanationText').innerText = step.explanation;
            document.getElementById('currentStepBadge').innerText = `Step ${idx + 1}/${stepList.length}`;

            if (idx === stepList.length - 1) {
                showResultBanner("Constructor & Inheritance Execution Completed", false);
                const btnStep = document.getElementById('btnStep');
                btnStep.disabled = true;
                btnStep.className = "bg-slate-700 text-slate-400 font-medium px-4 py-2 rounded-lg text-sm transition cursor-not-allowed";
                document.getElementById('currentStepBadge').innerText = "Finished";
            }
        }

        function stepExecute() {
            if (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function runAll() {
            startSimulation();
            while (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function resetSimulation() {
            currentStepIdx = 0;
            highlightLine(0);
            clearConsole();
            hideResultBanner();
            renderHeapObjects([]);
            appendConsole("<p class='text-slate-500 italic'>>>> Click 'Run / Execute All' or 'Start Simulation' to execute.</p>");
            document.getElementById('explanationText').innerText = "Select parameters and click 'Start Simulation' or 'Run / Execute All' to execute.";
            document.getElementById('currentStepBadge').innerText = "Step: Ready";

            const btnStep = document.getElementById('btnStep');
            btnStep.disabled = true;
            btnStep.className = "bg-slate-700 text-slate-400 font-medium px-4 py-2 rounded-lg text-sm transition cursor-not-allowed";
        }

                response = "Inheritance allows child classes to inherit attributes from parent classes (`class Child(Parent):`). Defining a method with the same name in the child class overrides the parent method!";

                const aiDiv = document.createElement('div');
                aiDiv.className = 'flex items-start gap-3';
