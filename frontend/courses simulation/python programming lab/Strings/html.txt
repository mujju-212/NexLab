            <h2 class="text-xl font-bold text-indigo-400 border-b border-slate-700 pb-2">Theory: Python Strings, Slicing & Immutability</h2>
            <p class="text-sm text-slate-300 leading-relaxed">
                In Python, a <strong>String</strong> (<code class="bg-slate-900 text-amber-300 px-1.5 py-0.5 rounded font-mono">str</code>) is an immutable sequence of Unicode characters enclosed in single quotes (<code class="text-emerald-300">'...'</code>), double quotes (<code class="text-emerald-300">"..."</code>), or triple quotes (<code class="text-emerald-300">"""..."""</code>). Strings support rich operations like indexing, slicing, concatenation, substring searching, and string-specific methods.
            </p>

            <!-- Grid of Concepts -->
            <div class="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4 text-sm">
                
                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-arrow-right-to-city text-emerald-400"></i> String Indexing</span>
                        <span class="text-[10px] bg-indigo-500/20 text-indigo-300 px-2 py-0.5 rounded font-sans">0-Based</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Characters are accessed using square brackets <code class="text-amber-300">s[i]</code>. Positive indices start from <code class="text-slate-200">0</code> (left to right), while negative indices start from <code class="text-slate-200">-1</code> (right to left).
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-scissors text-amber-400"></i> String Slicing</span>
                        <span class="text-[10px] bg-indigo-500/20 text-indigo-300 px-2 py-0.5 rounded font-sans">s[start:stop:step]</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Extracts substrings using <code class="text-amber-300">s[start:stop:step]</code>. <code class="text-slate-200">stop</code> is exclusive. E.g., <code class="text-emerald-300">"Python"[0:2]</code> yields <code class="text-emerald-300">"Py"</code>.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-lock text-rose-400"></i> String Immutability</span>
                        <span class="text-[10px] bg-rose-500/20 text-rose-300 px-2 py-0.5 rounded font-sans">Immutable</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        String characters cannot be changed in place! Attempting <code class="text-rose-400">s[0] = 'X'</code> raises a <code class="text-rose-400">TypeError</code>. Any string operation returns a new string.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-wand-magic-sparkles text-cyan-400"></i> String Methods</span>
                        <span class="text-[10px] bg-emerald-500/20 text-emerald-300 px-2 py-0.5 rounded font-sans">str Methods</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Built-in string transformations include <code class="text-amber-300">upper()</code>, <code class="text-amber-300">lower()</code>, <code class="text-amber-300">replace(old, new)</code>, <code class="text-amber-300">split(delim)</code>, and <code class="text-amber-300">strip()</code>.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-quote-left text-purple-400"></i> f-String Formatting</span>
                        <span class="text-[10px] bg-indigo-500/20 text-indigo-300 px-2 py-0.5 rounded font-sans">f"..."</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Formatted string literals allow embedding Python expressions directly inside curly braces: <code class="text-emerald-300">f"Hello {name}, score: {score}"</code>.
                    </p>
                </div>

                <div class="bg-slate-900 p-4 rounded-lg border border-slate-700 space-y-2">
                    <h3 class="font-bold text-indigo-300 font-mono flex items-center justify-between">
                        <span><i class="fa-solid fa-magnifying-glass text-emerald-400"></i> Substring Search</span>
                        <span class="text-[10px] bg-emerald-500/20 text-emerald-300 px-2 py-0.5 rounded font-sans">find() / in</span>
                    </h3>
                    <p class="text-slate-400 text-xs leading-relaxed">
                        Check substring presence using <code class="text-amber-300">'Py' in 'Python'</code> (returns <code class="text-emerald-400">True</code>) or find index using <code class="text-amber-300">s.find('th')</code>.
                    </p>
                </div>

            </div>

        function renderCharGrid(strVal, highlightedRange = []) {
            const grid = document.getElementById('charIndexGrid');
            const lenBadge = document.getElementById('strLenBadge');
            
            if (!strVal || strVal === '--') {

                const negIdx = i - n;

                        <span class="text-[9px] text-indigo-400">${negIdx}</span>
                    </div>
                `;
            }
            html += `</div>`;
            grid.innerHTML = html;

            const btnStep = document.getElementById('btnStep');
            btnStep.disabled = false;
            btnStep.className = "bg-emerald-600 hover:bg-emerald-500 text-white font-medium px-4 py-2 rounded-lg text-sm transition active:scale-95 shadow-lg shadow-emerald-600/20";

            appendConsole(`>>> Executing: string_demo.py [Scenario: ${scenario}]`);
            executeStepAtIndex(0);
        }

        function executeStepAtIndex(idx) {
            const step = stepList[idx];
            highlightLine(step.line);

            // Update Variable 1 Memory Card
            document.getElementById('lblVar1').innerText = step.v1.label;
            document.getElementById('valVar1').innerText = step.v1.val;
            document.getElementById('typeVar1').innerText = step.v1.type;
            document.getElementById('mutVar1').innerText = step.v1.mut;

            // Update Variable 2 Memory Card
            document.getElementById('lblVar2').innerText = step.v2.label;
            document.getElementById('valVar2').innerText = step.v2.val;
            document.getElementById('typeVar2').innerText = step.v2.type;
            document.getElementById('mutVar2').innerText = step.v2.mut;

            // Update Variable 3 Memory Card
            document.getElementById('lblVar3').innerText = step.v3.label;
            document.getElementById('valVar3').innerText = step.v3.val;
            document.getElementById('typeVar3').innerText = step.v3.type;
            document.getElementById('mutVar3').innerText = step.v3.mut;

            triggerMemUpdate('memVar1');
            triggerMemUpdate('memVar2');
            triggerMemUpdate('memVar3');

            renderCharGrid(step.mainStr, step.highlightRange);

            appendConsole(step.consoleMsg);
            document.getElementById('explanationText').innerText = step.explanation;
            document.getElementById('currentStepBadge').innerText = `Step ${idx + 1}/${stepList.length}`;

            if (idx === stepList.length - 1) {
                const isErr = scenario === 'immutability';
                showResultBanner(isErr ? "Execution Error: TypeError" : "Execution Completed Successfully", isErr);
                const btnStep = document.getElementById('btnStep');
                btnStep.disabled = true;
                btnStep.className = "bg-slate-700 text-slate-400 font-medium px-4 py-2 rounded-lg text-sm transition cursor-not-allowed";
                document.getElementById('currentStepBadge').innerText = "Finished";
            }
        }

        function stepExecute() {
            if (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function runAll() {
            startSimulation();
            while (currentStepIdx < stepList.length - 1) {
                currentStepIdx++;
                executeStepAtIndex(currentStepIdx);
            }
        }

        function resetSimulation() {
            currentStepIdx = 0;
            highlightLine(0);
            clearMemory();
            clearConsole();
            hideResultBanner();
            renderCharGrid('--', []);
            appendConsole("<p class='text-slate-500 italic'>>>> Click 'Run / Execute All' or 'Start Simulation' to execute.</p>");
            document.getElementById('explanationText').innerText = "Select parameters and click 'Start Simulation' or 'Run / Execute All' to execute.";
            document.getElementById('currentStepBadge').innerText = "Step: Ready";

            const btnStep = document.getElementById('btnStep');
            btnStep.disabled = true;
            btnStep.className = "bg-slate-700 text-slate-400 font-medium px-4 py-2 rounded-lg text-sm transition cursor-not-allowed";
        }

        function highlightLine(lineNum) {
            document.querySelectorAll('.code-line').forEach(el => el.classList.remove('active'));
            if (lineNum > 0) {
                const target = document.getElementById(`line${lineNum}`);
                if (target) target.classList.add('active');
            }
        }

        function clearMemory() {
            ['lblVar1', 'lblVar2', 'lblVar3'].forEach((id, i) => document.getElementById(id).innerText = `var${i+1}`);
            ['valVar1', 'valVar2', 'valVar3'].forEach(id => document.getElementById(id).innerText = '--');
            ['typeVar1', 'typeVar2', 'typeVar3'].forEach(id => document.getElementById(id).innerText = '--');
            ['mutVar1', 'mutVar2', 'mutVar3'].forEach(id => document.getElementById(id).innerText = '--');
        }

        function clearConsole() {
            document.getElementById('terminal').innerHTML = '';
        }

        function appendConsole(htmlContent) {
            const term = document.getElementById('terminal');
            const p = document.createElement('p');
            p.innerHTML = htmlContent;
            term.appendChild(p);
            term.scrollTop = term.scrollHeight;
        }

        function triggerMemUpdate(memId) {
            const el = document.getElementById(memId);
            if (!el) return;
            el.classList.add('updated');
            setTimeout(() => { el.classList.remove('updated'); }, 500);
        }

        function showResultBanner(text, isError) {
            const banner = document.getElementById('resultBanner');
            const bannerText = document.getElementById('resultBannerText');
            const bannerIcon = document.getElementById('bannerIcon');
            const bannerStatus = document.getElementById('bannerStatus');

            bannerText.innerText = text;
            if (isError) {
                bannerText.className = "text-lg font-mono font-bold text-rose-400";
                bannerIcon.className = "w-10 h-10 rounded-full bg-rose-500/20 text-rose-400 flex items-center justify-center text-lg font-bold";
                bannerIcon.innerHTML = `<i class="fa-solid fa-triangle-exclamation"></i>`;
                bannerStatus.innerText = "Status: Error";
                bannerStatus.className = "text-xs font-mono text-rose-400 bg-rose-950/80 px-3 py-1 rounded-md border border-rose-800";
            } else {
                bannerText.className = "text-lg font-mono font-bold text-emerald-300";
                bannerIcon.className = "w-10 h-10 rounded-full bg-emerald-500/20 text-emerald-400 flex items-center justify-center text-lg font-bold";
                bannerIcon.innerHTML = `<i class="fa-solid fa-square-check"></i>`;
                bannerStatus.innerText = "Completed";
                bannerStatus.className = "text-xs font-mono text-emerald-400 bg-slate-900/80 px-3 py-1 rounded-md border border-slate-700";
            }
            banner.classList.remove('hidden');
        }

        function hideResultBanner() {
            document.getElementById('resultBanner').classList.add('hidden');
        }

        // AI Assistant Query Handler
        function sendAiQuery() {
            const input = document.getElementById('aiQueryInput');
            const query = input.value.trim();
            if (!query) return;

            const chatBox = document.getElementById('aiChatBox');

            const userDiv = document.createElement('div');
            userDiv.className = 'flex items-start gap-3 justify-end';
            userDiv.innerHTML = `
                <div class="bg-indigo-600/30 border border-indigo-500/40 p-3 rounded-lg text-slate-200">
                    ${query}
                </div>
                <div class="w-7 h-7 rounded-full bg-slate-700 flex items-center justify-center text-slate-300 text-xs shrink-0">You</div>
            `;
            chatBox.appendChild(userDiv);
            input.value = '';

            let response = "Try selecting different scenarios in the simulator to inspect how Python handles string memory, indexing, and immutability!";
            const qLower = query.toLowerCase();
            if (qLower.includes('slice') || qLower.includes('start') || qLower.includes('stop')) {
                response = "String slicing uses `s[start:stop:step]`. Remember that `start` is INCLUSIVE while `stop` is EXCLUSIVE. Try the 'Indexing & Slicing' preset!";
            } else if (qLower.includes('immutable') || qLower.includes('change') || qLower.includes('error')) {
                response = "Strings in Python are IMMUTABLE. Once created in memory, individual characters cannot be modified. Methods like `.replace()` or `.upper()` return a NEW string copy!";
            } else if (qLower.includes('split') || qLower.includes('list') || qLower.includes('method')) {
                response = "The `.split(delim)` method converts a string into a list of substrings using a delimiter (like comma or space). Check out the 'String Methods' preset!";
            } else if (qLower.includes('negative') || qLower.includes('index')) {
                response = "Negative indices count backwards from the end of the string: `s[-1]` is the last character, `s[-2]` is second to last, down to `s[-len(s)]` for the first character!";
            }

            setTimeout(() => {
                const aiDiv = document.createElement('div');
                aiDiv.className = 'flex items-start gap-3';
                aiDiv.innerHTML = `
                    <div class="w-7 h-7 rounded-full bg-indigo-600 flex items-center justify-center text-white text-xs shrink-0">AI</div>
                    <div class="bg-slate-800 p-3 rounded-lg border border-slate-700 text-slate-200">
                        ${response}
                    </div>
                `;
                chatBox.appendChild(aiDiv);
                chatBox.scrollTop = chatBox.scrollHeight;
            }, 500);
        }
